-- File: schema.sql
-- SQLite schema for the Synergesis Glyph Archive database

CREATE TABLE IF NOT EXISTS glyph_archive (
    -- Core Glyph Properties (from glyphEngine.js)
    id TEXT PRIMARY KEY NOT NULL,
    timestamp REAL NOT NULL,           -- Store as UNIX timestamp (REAL in SQLite)
    polarité TEXT CHECK(polarité IN ('+', '-', '0', '±', '?')), -- Added '?' for unknown/error
    fréquence INTEGER CHECK(fréquence BETWEEN 0 AND 144), -- Allow 0 for error state?
    poids INTEGER CHECK(poids BETWEEN 0 AND 10),       -- Allow 0 for error state?
    alignement TEXT CHECK(alignement IN ('Celestial', 'Chthonic', 'Void', 'Harmonic', 'Elemental', 'Error', '?')), -- Added 'Error','?'
    tags TEXT,                         -- Store as JSON string array '["tag1", "tag2"]'
    sourceIds TEXT,                    -- Store as JSON string array '["id1", "id2"]'
    entropy_score REAL,                -- Can be NULL if not calculated

    -- Δ-TRANSLATE Output & Metrics
    natural_prompt TEXT,
    ai_instruction TEXT,
    semantic_yaml TEXT,
    resonance REAL,
    emergence INTEGER,
    stability TEXT CHECK(stability IN ('HIGH', 'MEDIUM', 'LOW', 'N/A')), -- N/A for error/uncalculated
    cluster_signature TEXT,            -- Heuristic cluster name

    -- System & Metadata
    status TEXT DEFAULT 'received' CHECK(status IN ('received', 'translating', 'translated', 'processing', 'archived', 'error_translate', 'error_process')),
    last_updated REAL,                 -- Store as UNIX timestamp
    archetype TEXT,                    -- Assigned semantic role (Seed, Mirror, Gate etc.)
    loop_source TEXT                   -- Tracks origin within the agent loop (e.g., 'ingestor', 'llm_response', 'feedback')
);

-- Optional: Indexes for faster querying
CREATE INDEX IF NOT EXISTS idx_glyph_timestamp ON glyph_archive (timestamp);
CREATE INDEX IF NOT EXISTS idx_glyph_alignement ON glyph_archive (alignement);
CREATE INDEX IF NOT EXISTS idx_glyph_status ON glyph_archive (status);
CREATE INDEX IF NOT EXISTS idx_glyph_resonance ON glyph_archive (resonance);
