def _fetch_glyphs_by_tag(tag: str, limit: Optional[int] = None) -> List[GlyphData]:
    """Simulates fetching glyphs with a specific tag from the database."""
    logger.debug(f"Fetching glyphs with tag '{tag}' (limit: {limit}).")
    candidates = [deepcopy(g) for g in MOCK_GLYPH_DB if tag in g.get('tags', [])]
    # Sort by timestamp (oldest first maybe for fusion?)
    candidates.sort(key=lambda g: g.get('timestamp', 0))
    if limit:
        return candidates[:limit]
    return candidates

def _update_glyph_status(glyph_id: str, new_status: str):
    """Simulates updating a glyph's status in the database."""
    if new_status not in STATUS_VALUES:
        logger.warning(f"Attempted to set invalid status '{new_status}' for {glyph_id}")
        return
    logger.debug(f"Updating status of glyph {glyph_id} to '{new_status}'.")
    for glyph in MOCK_GLYPH_DB:
        if glyph['id'] == glyph_id:
            glyph['status'] = new_status
            glyph['last_updated'] = time.time()
            break

def _get_glyph_by_id(glyph_id: str) -> Optional[GlyphData]:
     """Simulates fetching a single glyph by ID."""
     for glyph in MOCK_GLYPH_DB:
         if glyph['id'] == glyph_id:
             return deepcopy(glyph)
     return None

def task_run_llm_cycle_with_reverse_translation():
    """
    Conceptual task demonstrating selecting a glyph, prompting an LLM (simulated),
    and converting the response back into a glyph using ai_to_glyph.
    """
    logger.info("Task: Running LLM Cycle + Reverse Translation...")
    # 1. Select a glyph (simple simulation: pick latest)
    # In real system, use logic from LLMPromptGenerator class
    trigger_glyph = MOCK_GLYPH_DB[-1] if MOCK_GLYPH_DB else None
    if not trigger_glyph:
        logger.info("LLM Cycle: No glyphs available to generate prompt.")
        return

    # 2. Translate glyph to get prompt
    translation = translate_glyph(trigger_glyph) # Assumes translate_glyph exists
    prompt = translation.get("natural_prompt")
    if not prompt or "[ERROR]" in prompt or "[FAILED]" in prompt:
        logger.error(f"LLM Cycle: Failed to generate prompt for glyph {trigger_glyph['id']}.")
        _update_glyph_status(trigger_glyph['id'], 'error_translate')
        return

    # 3. Get response from LLM (Simulated)
    llm_response_data = _simulate_llm_response(prompt)

    # 4. Convert response to Glyph using ai_to_glyph
    try:
        response_glyph = response_to_glyph(
            response=llm_response_data,
            source_context=f"llm_response_to:{trigger_glyph['id']}"
        )
        logger.info(f"LLM Cycle: Converted LLM response to glyph {response_glyph['id']}")
        # 5. Add the new glyph back into the system
        _add_glyph_to_system(response_glyph)

    except Exception as e:
        logger.error(f"LLM Cycle: Failed to convert LLM response to glyph: {e}", exc_info=True)
        # Optionally create an 'error' glyph here

             'id': f"error-resp-{int(time.time())}", 'timestamp': time.time(),

             'sourceIds': [f"error_handler:{trigger_glyph['id']}"],
             'details': {'error_message': str(e)}
        }
        _add_glyph_to_system(error_glyph) # type: ignore


def task_check_and_fuse_glyphs():
    """
    Checks for glyphs marked as fusion candidates and fuses them if threshold is met.
    """
    logger.info("Task: Checking for fusion candidates...")
    fusion_tag = CONFIG['fusion']['trigger_tag']
    threshold = CONFIG['fusion']['trigger_threshold']
    max_parents = CONFIG['fusion'].get('max_parents_per_fusion', 5) # Get max parents limit

    # Fetch candidates (limit query if possible in real DB)
    candidates = _fetch_glyphs_by_tag(fusion_tag)

    if len(candidates) >= threshold:
        logger.info(f"Found {len(candidates)} fusion candidates (>= threshold {threshold}). Initiating fusion.")

        # Select parents for this fusion run (e.g., oldest ones up to max_parents)
        parents_to_fuse = candidates[:max_parents]
        parent_ids = [g['id'] for g in parents_to_fuse]
        logger.info(f"Selected {len(parents_to_fuse)} parents for fusion: {parent_ids}")

        # --- Detailed Logging Setup ---
        # Calculate average metrics of parents BEFORE fusion for delta logging
        avg_parent_metrics = {'resonance': 0.0, 'emergence': 0.0, 'entropy': 0.0, 'count': 0}
        valid_metrics_count = {'resonance': 0, 'emergence': 0, 'entropy': 0}
        for p in parents_to_fuse:
            if pd.notna(p.get('resonance')):
                avg_parent_metrics['resonance'] += p['resonance'] # type: ignore
                valid_metrics_count['resonance'] += 1
            if pd.notna(p.get('emergence')):
                 avg_parent_metrics['emergence'] += p['emergence'] # type: ignore
                 valid_metrics_count['emergence'] += 1
            if pd.notna(p.get('entropy_score')):
                 avg_parent_metrics['entropy'] += p['entropy_score'] # type: ignore
                 valid_metrics_count['entropy'] += 1
        if valid_metrics_count['resonance'] > 0: avg_parent_metrics['resonance'] /= valid_metrics_count['resonance']
        if valid_metrics_count['emergence'] > 0: avg_parent_metrics['emergence'] /= valid_metrics_count['emergence']
        if valid_metrics_count['entropy'] > 0: avg_parent_metrics['entropy'] /= valid_metrics_count['entropy']
        avg_parent_metrics['count'] = len(parents_to_fuse)

        try:
            # Perform fusion using the engine
            fused_glyph = fuse_glyphs(
                glyph_list=parents_to_fuse,
                custom_strategies=CONFIG['fusion']['default_strategies']
            )

            # --- Log Fusion Details ---
            fusion_details = fused_glyph.get('details', {})
            strategy_log = fusion_details.get('fusion_strategy_used', 'N/A')
            logger.info(f"Fusion successful: {fused_glyph['id']} created from {len(parent_ids)} parents.")
            logger.debug(f"  Parents: {parent_ids}")
            logger.debug(f"  Strategy: {strategy_log}")

            # Log metric deltas
            fused_res = fused_glyph.get('resonance') # Fused glyph might not have metrics yet
            fused_emg = fused_glyph.get('emergence')
            fused_ent = fused_glyph.get('entropy_score')
            if pd.notna(fused_res): logger.info(f"  ΔResonance: {fused_res - avg_parent_metrics['resonance']:.2f} (Fused: {fused_res:.2f}, Avg Parent: {avg_parent_metrics['resonance']:.2f})") # type: ignore
            if pd.notna(fused_emg): logger.info(f"  ΔEmergence: {fused_emg - avg_parent_metrics['emergence']:.0f} (Fused: {fused_emg:.0f}, Avg Parent: {avg_parent_metrics['emergence']:.0f})") # type: ignore
            if pd.notna(fused_ent): logger.info(f"  ΔEntropy: {fused_ent - avg_parent_metrics['entropy']:.3f} (Fused: {fused_ent:.3f}, Avg Parent: {avg_parent_metrics['entropy']:.3f})") # type: ignore

            # Add the fused glyph back into the system
            _add_glyph_to_system(fused_glyph)

            # Update status of parent glyphs (e.g., 'fused_into:new_id')
            for parent_id in parent_ids:
                _update_glyph_status(parent_id, f"fused_into:{fused_glyph['id']}")

        except ValueError as ve:
            logger.error(f"Fusion failed: {ve}. Parents: {parent_ids}")
        except Exception as e:
            logger.error(f"Unexpected error during fusion: {e}. Parents: {parent_ids}", exc_info=True)
            # Update parent status to error?
            for parent_id in parent_ids:
                 _update_glyph_status(parent_id, 'error_fusion')
    else:
        logger.debug(f"Found {len(candidates)} fusion candidates. Threshold ({threshold}) not met.")

# --- Main Loop ---
def main_loop():
    logger.info("--- Starting Synergesis Agent Loop (Phase IV) ---")
    setup_schedule()
    logger.info(f"Entering main loop. Loop sleep: {CONFIG['general']['loop_sleep_seconds']}s")
    try:
        while True:
            schedule.run_pending()
            # Check idle time to avoid busy-waiting
            idle_seconds = schedule.idle_seconds()
            sleep_duration = CONFIG['general']['loop_sleep_seconds']
            if idle_seconds is not None and idle_seconds > 0:
                 # Sleep only as long as needed, up to the configured max
                 sleep_duration = min(sleep_duration, idle_seconds)

            # logger.debug(f"Sleeping for {sleep_duration:.2f} seconds...")
            time.sleep(max(0.1, sleep_duration)) # Ensure minimum sleep to prevent high CPU

    except KeyboardInterrupt:
        logger.info("KeyboardInterrupt received. Shutting down loop...")
    except Exception as e:
        logger.critical(f"Critical error in main loop: {e}", exc_info=True)
    finally:
        logger.info("--- Synergesis Agent Loop Shutdown ---")

Explanation and Integration Points:

Imports: The script now imports GlyphData from glyph_core, fuse_glyphs from glyph_fusion_engine, and response_to_glyph from ai_to_glyph. It also imports translate_glyph conceptually for the LLM cycle.

Mock System: Placeholder functions (_add_glyph_to_system, _fetch_glyphs_by_tag, _update_glyph_status, _simulate_llm_response) simulate database/queue interactions and LLM calls using an in-memory list (MOCK_GLYPH_DB). Replace these with actual database/API calls in your full system.

Fusion Task (task_check_and_fuse_glyphs):

Fetches glyphs tagged with fuse_candidate (configurable).

Checks if the count meets the trigger_threshold.

Selects a limited number of parents (max_parents_per_fusion).

Calculates average parent metrics before fusion.

Calls fuse_glyphs, passing the selected parents and configured strategies.

Logs parent IDs, new ID, strategy used, and metric deltas (ΔResonance, ΔEmergence, ΔEntropy).

Adds the fused_glyph back to the system.

Updates the status of parent glyphs.

LLM Task (task_run_llm_cycle_with_reverse_translation):

Simulates selecting a glyph and generating a prompt via translate_glyph.

Calls the simulated LLM.

Uses response_to_glyph to convert the result (string or dict) into a new GlyphData object.

Adds the response_glyph back into the system.

Scheduling: The new task_check_and_fuse_glyphs and the updated task_run_llm_cycle_with_reverse_translation are added to the schedule.

Dev Console (run_dev_console):

Provides a simple command-line interface triggered when the script is run directly (if __name__ == "__main__":).

Allows direct testing of fuse_glyphs (test_fusion) and response_to_glyph (test_reverse) with sample/randomized data.
