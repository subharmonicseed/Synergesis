def get_dominant_property(prop_value: Union[str, List, Dict, None], prop_type: str) -> str:
    """Gets the dominant simple value from potentially composite properties."""
    if prop_type == 'polarity':
        if isinstance(prop_value, list) and prop_value:
            # Return char with highest influence
            return max(prop_value, key=lambda item: item[1])[0]
        elif isinstance(prop_value, str):
            return prop_value
        else: return '?'
    elif prop_type == 'alignment':
         if isinstance(prop_value, dict) and prop_value:
             # Return key with highest influence
             return max(prop_value, key=prop_value.get)
         elif isinstance(prop_value, str):
             return prop_value
         else: return 'Error'
    return '?'

# @st.cache_data # Caching filter results can improve performance
def filter_glyphs(glyphs: List[GlyphData], filters: Dict[str, Any]) -> List[GlyphData]:
    """Filters the list of glyphs based on provided criteria."""
    filtered = glyphs
    if filters.get('polarities'):
        allowed_pols = set(filters['polarities'])
        filtered = [g for g in filtered if get_dominant_property(g.get('polarité'), 'polarity') in allowed_pols]
    if filters.get('alignments'):
        allowed_alns = set(filters['alignments'])
        filtered = [g for g in filtered if get_dominant_property(g.get('alignement'), 'alignment') in allowed_alns]
    if filters.get('tags'):
        search_tags = {t.strip().lower() for t in filters['tags'].split(',') if t.strip()}
        if search_tags:
             filtered = [g for g in filtered if search_tags.issubset(set(tg.lower() for tg in g.get('tags', [])))]
    if filters.get('status'):
        allowed_stats = set(filters['status'])
        filtered = [g for g in filtered if g.get('status') in allowed_stats]
    if filters.get('id_search'):
        search_id = filters['id_search'].lower()
        filtered = [g for g in filtered if search_id in g.get('id', '').lower()]

    return filtered

def render_glyph_svg(glyph: GlyphData, size=200) -> str:
    """Generates an SVG visualization of a single glyph based on properties."""
    # --- Get Dominant Properties for Base Shape/Color ---
    dom_polarity = get_dominant_property(glyph.get('polarité'), 'polarity')
    dom_alignment = get_dominant_property(glyph.get('alignement'), 'alignment')
    shape_type, pol_color = POLARITY_SHAPES.get(dom_polarity, ('diamond', 'grey'))
    aln_color = ALIGNMENT_COLORS.get(dom_alignment, 'grey')

    # --- Map Metrics to Visuals ---
    # Resonance -> Radius (scaled non-linearly maybe?)
    resonance = glyph.get('resonance', 50.0) or 50.0 # Default if None
    base_radius = 15 + math.sqrt(max(0, resonance)) * 3.5 # Scale sqrt to prevent huge sizes
    radius = max(5, min(48, base_radius)) # Clamp radius to viewbox limits

    # Entropy -> Opacity (0=opaque, 1=transparent)
    entropy = glyph.get('entropy_score', 0.5) or 0.5
    opacity = round(max(0.1, min(1.0, 1.0 - entropy * 0.8)), 2) # Higher entropy = more transparent

    # Stability -> Border
    stability = glyph.get('stability', 'MEDIUM')
    border_dash, border_width = STABILITY_BORDERS.get(stability, ("solid", 1.5))
    # Make border color slightly darker/contrasting alignment color
    border_color = pol_color # Use polarity color for border

    # --- Generate SVG ---
    svg = f'<svg width="{size}" height="{size}" viewBox="0 0 100 100" style="background-color: #f0f0f0; border-radius: 5px;">'
    svg += f'<title>Glyph ID: {glyph["id"]}\nRes: {resonance:.1f}, Ent: {entropy:.2f}, Stab: {stability}</title>' # Tooltip

    shape_props = f'fill="{aln_color}" fill-opacity="{opacity}" stroke="{border_color}" stroke-width="{border_width}"'
    if border_dash != "solid":
        shape_props += f' stroke-dasharray="{border_dash}"'

    # Draw shape based on polarity
    cx, cy = 50, 50 # Center
    if shape_type == 'circle':
        svg += f'<circle cx="{cx}" cy="{cy}" r="{radius}" {shape_props} />'
    elif shape_type == 'square':
        s = radius * 1.8 # side length
        svg += f'<rect x="{cx-s/2}" y="{cy-s/2}" width="{s}" height="{s}" {shape_props} />'
    elif shape_type == 'triangle': # Pointing up
        h = radius * 1.732 # height based on radius roughly
        w = radius * 2 # base width
        svg += f'<polygon points="{cx},{cy-h*0.6} {cx+w/2},{cy+h*0.4} {cx-w/2},{cy+h*0.4}" {shape_props} />'
    elif shape_type == 'diamond':
        w = radius * 1.5
        h = radius * 2
        svg += f'<polygon points="{cx},{cy-h/2} {cx+w/2},{cy} {cx},{cy+h/2} {cx-w/2},{cy}" {shape_props} />'
    elif shape_type == 'star': # 5-point star
         outer_r = radius
         inner_r = radius * 0.45
         points = []
         for i in range(5):
             angle_outer = math.pi / 2 - 2 * math.pi * i / 5
             angle_inner = math.pi / 2 - 2 * math.pi * (i + 0.5) / 5
             points.append(f"{cx + outer_r * math.cos(angle_outer)},{cy - outer_r * math.sin(angle_outer)}")
             points.append(f"{cx + inner_r * math.cos(angle_inner)},{cy - inner_r * math.sin(angle_inner)}")
         svg += f'<polygon points="{" ".join(points)}" {shape_props} />'

    # Add Frequency/Weight text indicator?
    freq = glyph.get('fréquence', '?')
    weight = glyph.get('poids', '?')
    svg += f'<text x="5" y="15" font-family="monospace" font-size="10" fill="#333">F:{freq}</text>'
    svg += f'<text x="5" y="90" font-family="monospace" font-size="10" fill="#333">W:{weight}</text>'

    svg += '</svg>'
    return svg

def compute_coherence_metrics(glyphs: List[GlyphData]) -> Dict[str, Any]:
    """Computes overall system coherence metrics from the glyph list."""
    if not glyphs:
        return {"total": 0, "avg_res": 0, "avg_ent": 0.5, "avg_emg": 0, "align_dist": {}, "status_dist": {}, "fused_count": 0}

    df = pd.DataFrame(glyphs)
    df['timestamp_dt'] = pd.to_datetime(df['timestamp'], unit='s', utc=True)
    df = df.set_index('timestamp_dt')

    metrics = {
        "total": len(df),
        "avg_res": pd.to_numeric(df['resonance'], errors='coerce').mean(),
        "avg_ent": pd.to_numeric(df['entropy_score'], errors='coerce').mean(),
        "avg_emg": pd.to_numeric(df['emergence'], errors='coerce').mean(),
        # Get dominant alignment for distribution
        "align_dist": df['alignement'].apply(lambda x: get_dominant_property(x, 'alignment')).value_counts().to_dict(),
        "status_dist": df['status'].fillna('unknown').value_counts().to_dict(),
        "fused_count": df['tags'].apply(lambda t: 'fused' in t if isinstance(t, list) else False).sum()
    }
    # Fill NaN with 0 or appropriate defaults
    metrics["avg_res"] = metrics["avg_res"] if pd.notna(metrics["avg_res"]) else 0.0
    metrics["avg_ent"] = metrics["avg_ent"] if pd.notna(metrics["avg_ent"]) else 0.5
    metrics["avg_emg"] = metrics["avg_emg"] if pd.notna(metrics["avg_emg"]) else 0.0

    return metrics

def plot_metrics(glyphs: List[GlyphData]):
    """Generates and displays coherence charts using Plotly."""
    if len(glyphs) < 2:
        st.warning("Insufficient data for plotting metrics (need at least 2 glyphs).")
        return

    try:
        import plotly.express as px
        import plotly.graph_objects as go

        df = pd.DataFrame(glyphs)
        df['timestamp_dt'] = pd.to_datetime(df['timestamp'], unit='s', utc=True)
        # Ensure numeric types, coercing errors
        numeric_cols = ['resonance', 'entropy_score', 'emergence']
        for col in numeric_cols:
            df[col] = pd.to_numeric(df[col], errors='coerce')
        df = df.sort_values('timestamp_dt')

        st.subheader("Metric Trends Over Time")
        # Simple line chart for core metrics
        df_melted = df.melt(id_vars='timestamp_dt', value_vars=numeric_cols, var_name='Metric', value_name='Value')
        fig_trends = px.line(df_melted, x='timestamp_dt', y='Value', color='Metric',
                             title="Resonance, Entropy & Emergence Over Time", markers=True)
        fig_trends.update_layout(legend_title_text='Metric')
        st.plotly_chart(fig_trends, use_container_width=True)

        st.subheader("Current Distribution")
        col_pie, col_bar = st.columns(2)
        with col_pie:
            # Alignment Distribution

# --- Tab 4: Fusion Engine ---
with tab_fusion:
    st.header("Symbolic Fusion Engine")
    st.markdown("Select glyphs tagged for fusion and combine them.")

    fusion_candidates = filter_glyphs(all_glyphs, {"tags": FUSION_CANDIDATE_TAG})

    if not fusion_candidates:
        st.info(f"No glyphs found with the tag '{FUSION_CANDIDATE_TAG}'.")
    else:
        st.write(f"Found {len(fusion_candidates)} potential fusion candidates:")

        # Use checkboxes within a form for selection
        with st.form("fusion_form"):
            selected_indices = []
            for i, glyph in enumerate(fusion_candidates):
                is_selected = st.checkbox(f"ID: `{glyph['id'][:12]}...` (P:{get_dominant_property(glyph.get('polarité'), 'polarity')}, A:{get_dominant_property(glyph.get('alignement'), 'alignment')}, W:{glyph.get('poids')})", key=f"fuse_sel_{glyph['id']}")
                if is_selected:
                    selected_indices.append(i)

            # Fusion button within the form
            submitted = st.form_submit_button("⚛️ Fuse Selected Glyphs")

            if submitted:
                if len(selected_indices) < 2:
                    st.warning("Please select at least two glyphs to fuse.")
                else:
                    glyphs_to_fuse_dicts = [fusion_candidates[i] for i in selected_indices]
                    parent_ids = [g['id'] for g in glyphs_to_fuse_dicts]
                    st.info(f"Attempting to fuse {len(glyphs_to_fuse_dicts)} glyphs: {parent_ids}")
                    try:
                        # --- Trigger Fusion ---
                        fused_glyph = fuse_glyphs(glyphs_to_fuse_dicts) # Call the engine function

                        # --- IMPORTANT: Update the central 'database' ---
                        # 1. Add the new fused glyph
                        st.session_state.mock_glyph_db.append(fused_glyph)
                        # 2. Update status of parents (find and modify in the list)
                        new_status = f"fused_into:{fused_glyph['id']}"
                        if new_status not in STATUS_VALUES: STATUS_VALUES.append(new_status) # Add dynamic status
                        updated_count = 0
                        for i, g in enumerate(st.session_state.mock_glyph_db):
                            if g['id'] in parent_ids:
                                st.session_state.mock_glyph_db[i]['status'] = new_status
                                st.session_state.mock_glyph_db[i]['last_updated'] = time.time()
                                updated_count += 1

                        st.success(f"Fusion successful! New glyph '{fused_glyph['id']}' created. Updated status for {updated_count} parents.")
                        st.subheader("Fused Glyph Details:")
                        st.json(fused_glyph, expanded=True)
                        # Clear selection? Might be good UX.
                        # st.session_state.fusion_selection = [] # This state wasn't used, handled by form state
                        # Trigger rerun to refresh lists/metrics
                        st.experimental_rerun()

                    except Exception as e:
                        logger.error(f"Error during dashboard fusion trigger: {e}", exc_info=True)
                        st.error(f"Fusion failed: {e}")
