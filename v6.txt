        metrics["polarity_dist"] = df['dom_pol'].value_counts(normalize=True).to_dict()

        # Error Rate
        metrics["error_rate"] = metrics["alignment_dist"].get("Error", 0.0) + \
                                df['tags'].apply(lambda t: isinstance(t, list) and any(err_tag in t for err_tag in ['error', 'failed'])).mean() * 0.5 # Add tag checks

        # Handle NaNs
        for key in ["avg_resonance", "avg_entropy", "avg_stability_score"]:
            if pd.isna(metrics[key]): metrics[key] = None

        return metrics

def render_glyph_svg(glyph: GlyphData, size=200, highlight_style: Optional[str]=None) -> str:
    """Generates an SVG visualization, adding optional highlight effects."""
    # ... (Keep existing logic for shape, color, opacity, border from Phase V) ...
    dom_polarity = get_dominant_property(glyph.get('polarité'), 'polarity')
    dom_alignment = get_dominant_property(glyph.get('alignement'), 'alignment')
    shape_type, pol_color = POLARITY_SHAPES.get(dom_polarity, ('diamond', 'grey'))
    aln_color = ALIGNMENT_COLORS.get(dom_alignment, 'grey')
    resonance = glyph.get('resonance', 50.0) or 50.0
    base_radius = 15 + math.sqrt(max(0, resonance)) * 3.5
    radius = max(5, min(48, base_radius))
    entropy = glyph.get('entropy_score', 0.5) or 0.5
    opacity = round(max(0.1, min(1.0, 1.0 - entropy * 0.8)), 2)
    stability = glyph.get('stability', 'MEDIUM')
    border_dash, border_width = STABILITY_BORDERS.get(stability, ("solid", 1.5))
    border_color = pol_color

    svg = f'<svg width="{size}" height="{size}" viewBox="0 0 100 100" style="background-color: #f0f0f0; border-radius: 5px;">'
    svg += f'<title>Glyph ID: {glyph["id"]}\nRes: {resonance:.1f}, Ent: {entropy:.2f}, Stab: {stability}</title>'

    # --- Add Highlight Effect ---
    if highlight_style == 'reflexive_action': # e.g., for cortex generated glyphs
         # Pulsing glow effect using animation
         glow_color = "#facc15" # Yellow/Gold glow
         svg += f"""
            <defs>
                <filter id="glow_reflex" x="-50%" y="-50%" width="200%" height="200%">
                    <feGaussianBlur stdDeviation="4" result="coloredBlur"/>
                    <feMerge>
                        <feMergeNode in="coloredBlur"/>
                        <feMergeNode in="SourceGraphic"/>
                    </feMerge>
                </filter>
            </defs>
            <circle cx="50" cy="50" r="{radius + 5}" fill="{glow_color}" filter="url(#glow_reflex)">
                 <animate attributeName="opacity" values="0.5;1;0.5" dur="2s" repeatCount="indefinite"/>
            </circle>
         """
    elif highlight_style == 'diagnosis_target': # e.g., glyph diagnosed as problematic
        # Simple red outline/shadow
        svg += f'<circle cx="50" cy="50" r="{radius + 3}" fill="none" stroke="#e11d48" stroke-width="3" opacity="0.7"/>'
    # --- End Highlight Effect ---

    # --- Draw main shape (same as before) ---
    shape_props = f'fill="{aln_color}" fill-opacity="{opacity}" stroke="{border_color}" stroke-width="{border_width}"'
    if border_dash != "solid": shape_props += f' stroke-dasharray="{border_dash}"'
    cx, cy = 50, 50
    # ... (Draw circle, square, triangle, etc. based on shape_type) ...
    # Example for circle:
    if shape_type == 'circle': svg += f'<circle cx="{cx}" cy="{cy}" r="{radius}" {shape_props} />'
    # ... (Add other shapes) ...
    else: svg += f'<rect x="{cx-radius}" y="{cy-radius}" width="{radius*2}" height="{radius*2}" {shape_props} />' # Fallback rect


    # --- Add Text indicators (same as before) ---
    freq = glyph.get('fréquence', '?')
    weight = glyph.get('poids', '?')
    svg += f'<text x="5" y="15" font-family="monospace" font-size="10" fill="#333">F:{freq}</text>'
    svg += f'<text x="5" y="90" font-family="monospace" font-size="10" fill="#333">W:{weight}</text>'
    svg += '</svg>'
    return svg
