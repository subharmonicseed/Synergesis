/**
 * Generates a random floating-point number within a specified range.
 * @param {number} min - The minimum value (inclusive).
 * @param {number} max - The maximum value (exclusive).
 * @returns {number} A random number between min and max.
 */
function randomRange(min, max) {
    return Math.random() * (max - min) + min;
}

/**
 * Generates a random integer within a specified range (inclusive).
 * @param {number} min - The minimum integer value.
 * @param {number} max - The maximum integer value.
 * @returns {number} A random integer between min and max.
 */
function randomInt(min, max) {
    min = Math.ceil(min);
    max = Math.floor(max);
    return Math.floor(Math.random() * (max - min + 1)) + min;
}

/**
 * Selects a random element from an array.
 * @param {Array<*>} arr - The array to choose from.
 * @returns {*} A random element from the array, or undefined if the array is empty.
 */
function randomChoice(arr) {
    if (!arr || arr.length === 0) return undefined;
    return arr[Math.floor(Math.random() * arr.length)];
}

/**
 * Generates a simple timestamp-based unique ID. Sufficient for internal use.
 * Not cryptographically secure or guaranteed globally unique like UUID v4.
 * @returns {string} A unique identifier string.
 */
function generateUUID() {
    // Simple implementation for internal tool uniqueness
    return `syn-${Date.now().toString(36)}-${Math.random().toString(36).substring(2, 9)}`;
}

/**
 * Performs a deep clone of an object using JSON stringify/parse.
 * Works for simple objects, arrays, and primitives. Loses functions, Dates become strings, etc.
 * Sufficient for cloning glyph data for archiving.
 * @param {object} obj - The object to clone.
 * @returns {object|null} The cloned object, or null if cloning fails.
 */
function deepClone(obj) {
    if (obj === null || typeof obj !== 'object') {
        return obj; // Return primitives directly
    }
    try {
        // Handle potential Date objects (convert back after parse)
        // This basic version doesn't handle Dates, Functions, Regexps, Maps, Sets correctly.
        // For this tool's data structure (simple objects, arrays, strings, numbers), it's okay.
        return JSON.parse(JSON.stringify(obj));
    } catch (e) {
        console.error("DeepClone Error:", e, "Object:", obj);
        return null; // Indicate failure
    }
}

/**
 * Normalizes a value to a range between 0 and 1.
 * @param {number} value - The value to normalize.
 * @param {number} min - The minimum possible value in the original range.
 * @param {number} max - The maximum possible value in the original range.
 * @returns {number} The normalized value (clamped between 0 and 1).
 */
function normalize(value, min, max) {
    if (max === min) return 0.5; // Avoid division by zero, return neutral value
    return Math.max(0, Math.min(1, (value - min) / (max - min)));
}

/**
 * Calculates the mouse position relative to the canvas element,
 * accounting for CSS scaling and border/padding.
 * @param {HTMLCanvasElement} canvas - The canvas element.
 * @param {MouseEvent} event - The mouse event.
 * @returns {{x: number, y: number}} The coordinates {x, y} relative to the canvas's internal resolution.
 */
function getCanvasMousePos(canvas, event) {
     const rect = canvas.getBoundingClientRect();
     // Calculate scale based on actual rendered size vs internal resolution
     const scaleX = canvas.width / rect.width;
     const scaleY = canvas.height / rect.height;
     // Calculate mouse position relative to the canvas top-left
     const canvasX = (event.clientX - rect.left) * scaleX;
     const canvasY = (event.clientY - rect.top) * scaleY;
     return { x: canvasX, y: canvasY };
}

/**
 * Draws a single geometric shape on the canvas context.
 * Assumes context transformations (translate) are handled externally for glyph positioning.
 * Positions within the shape object are relative to the glyph's center (0,0 after translate).
 * @param {CanvasRenderingContext2D} ctx - The canvas rendering context.
 * @param {object} shape - The shape object containing type and geometric properties.
 * @param {string} shape.type - Type of shape ('circle', 'triangle', 'line', 'arc').
 * @param {number} shape.x - Relative X position.
 * @param {number} shape.y - Relative Y position.
 * @param {number} [shape.radius] - Radius (for circle, arc).
 * @param {number} [shape.size] - Size parameter (e.g., base for triangle).
 * @param {number} [shape.angle] - Rotation angle in radians (for triangle, line).
 * @param {number} [shape.length] - Length (for line).
 * @param {number} [shape.startAngle] - Start angle in radians (for arc).
 * @param {number} [shape.endAngle] - End angle in radians (for arc).
 */
function drawShape(ctx, shape) {
    ctx.beginPath();
    const x = shape.x; // Positions are already relative
    const y = shape.y;

    switch (shape.type) {
        case 'circle':
            if (typeof shape.radius !== 'number') { console.warn("Missing radius for circle:", shape); return; }
            ctx.arc(x, y, shape.radius, 0, Math.PI * 2);
            break;
        case 'triangle':
             if (typeof shape.size !== 'number' || typeof shape.angle !== 'number') { console.warn("Missing size/angle for triangle:", shape); return; }
             // Equilateral triangle for simplicity
            ctx.save();
            ctx.translate(x, y);
            ctx.rotate(shape.angle);
            const h = shape.size * (Math.sqrt(3) / 2); // Height
            ctx.moveTo(0, -h / 2); // Top point
            ctx.lineTo(-shape.size / 2, h / 2); // Bottom left
            ctx.lineTo(shape.size / 2, h / 2); // Bottom right
            ctx.closePath();
            ctx.restore();
            break;
        case 'line':
             if (typeof shape.length !== 'number' || typeof shape.angle !== 'number') { console.warn("Missing length/angle for line:", shape); return; }
             // Draw line centered around (x, y) with given length and angle
             const dx = Math.cos(shape.angle) * shape.length / 2;
             const dy = Math.sin(shape.angle) * shape.length / 2;
             ctx.moveTo(x - dx, y - dy);
             ctx.lineTo(x + dx, y + dy);
            break;
        case 'arc':
            if (typeof shape.radius !== 'number' || typeof shape.startAngle !== 'number' || typeof shape.endAngle !== 'number') { console.warn("Missing radius/angles for arc:", shape); return; }
            ctx.arc(x, y, shape.radius, shape.startAngle, shape.endAngle);
            break;
        default:
            console.warn(`Attempted to draw unknown shape type: ${shape.type}`, shape);
            return; // Don't stroke unknown shapes
    }
    ctx.stroke(); // Apply the stroke defined before calling this function
}

/**
 * Simple text wrapping function for drawing text within a constrained width.
 * Used primarily if drawing analysis directly onto canvas was needed (not currently used for DOM panel).
 * @param {CanvasRenderingContext2D} ctx - Canvas context.
 * @param {string} text - Text to wrap.
 * @param {number} x - Starting X position.
 * @param {number} y - Starting Y position.
 * @param {number} maxWidth - Maximum width for a line.
 * @param {number} lineHeight - Height of each line.
 */
function wrapText(ctx, text, x, y, maxWidth, lineHeight) {
    const words = text.split(' ');
    let line = '';
    for(let n = 0; n < words.length; n++) {
      const testLine = line + words[n] + ' ';
      const metrics = ctx.measureText(testLine);
      const testWidth = metrics.width;
      if (testWidth > maxWidth && n > 0) {
        ctx.fillText(line, x, y);
        line = words[n] + ' ';
        y += lineHeight;
      } else {
        line = testLine;
      }
    }
    ctx.fillText(line, x, y);
}
