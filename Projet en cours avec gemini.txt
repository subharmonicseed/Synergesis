def _get_map_value(value: int, mapping: Dict[Tuple[int, int], Dict], default_key_index: int = 0) -> Dict:
    """Helper to find the correct dictionary from range-based maps."""
    for (min_v, max_v), result in mapping.items():
        if min_v <= value <= max_v:
            return result
    keys = list(mapping.keys())
    return mapping[keys[default_key_index]] if keys else {}

def _format_tags(tags: Optional[List[str]]) -> Tuple[str, str]:
    """Formats tag list for NL and YAML."""
    if not tags: return "the core subject", "[]"
    tags_clean = [str(t).replace("'", "’").strip() for t in tags if t and isinstance(t, str)] # Clean tags
    if not tags_clean: return "the core subject", "[]"
    if len(tags_clean) == 1: return f"the concept of '{tags_clean[0]}'", f"['{tags_clean[0]}']"
    if len(tags_clean) == 2: return f"the concepts '{tags_clean[0]}' and '{tags_clean[1]}'", f"['{tags_clean[0]}', '{tags_clean[1]}']"
    nl_tags = ", ".join([f"'{t}'" for t in tags_clean[:-1]]) + f", and '{tags_clean[-1]}'"
    yaml_tags = '[' + ', '.join([f"'{t}'" for t in tags_clean]) + ']'
    return f"the concepts {nl_tags}", yaml_tags

def _normalize(value: Optional[float], min_val: float, max_val: float) -> float:
    """Normalizes a value between 0 and 1."""
    if pd.isna(value) or pd.isna(min_val) or pd.isna(max_val) or max_val == min_val: return 0.5
    return max(0.0, min(1.0, (float(value) - min_val) / (max_val - min_val)))

class SimpleNLPEngine:
    def analyze(self, text: str) -> Dict[str, Any]:
        text_l = text.lower()
        sentiment = 0.0
        # Simple keyword spotting for sentiment
        pos_words = ['good','great','excellent','positive','success','create','generate','up','increase','profit']
        neg_words = ['bad','poor','negative','fail','error','down','decrease','loss','chaos','stuck']
        if any(w in text_l for w in pos_words): sentiment += 0.6
        if any(w in text_l for w in neg_words): sentiment -= 0.6
        sentiment = max(-1.0, min(1.0, sentiment))
        # Extract longer words as potential keywords
        keywords = [w for w in re.findall(r'\b\w{4,}\b', text_l)][:5]
        logger.debug(f"Simple NLP: Text='{text_l[:30]}...' -> Sent={sentiment:.2f}, KW={keywords}")
