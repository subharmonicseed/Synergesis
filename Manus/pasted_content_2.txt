"""Synergesis – Vyra Agent
================================
Vyra est l’agent « créateur » chargé d’identifier des lacunes dans NOUS et de
proposer de nouveaux concepts symboliques à partir de glyphs, d’observations ou
d’instructions internes.  Il s’interface :

• avec le bus interne (glyph_bus) pour recevoir des glyphs ou messages
  `knowledge_gap`,
• avec le module NOUS via son API REST (par défaut http://localhost:8001),
• avec un LLM (OpenAI GPT par défaut, via la variable d’environnement
  `OPENAI_API_KEY`).

Points clefs
------------
- **BaseAgent** : on hérite de synergesis.agents.core_agents.BaseAgent afin
  d’être planifiable via agent_loop.
- **NousClient** : wrapper minimal `get_concept`, `search_concepts`,
  `add_concept`.
- **LLMBackend** : adaptatif ; par défaut OpenAI `gpt-4o-mini` ; fallback stub
  si la clé n’est pas définie.
- **Vyra.tick()** : consulte la queue interne `self.inbox` (alimentée par le
  bus), génère des concepts si besoin, et les poste dans NOUS.

"""
from __future__ import annotations

import os
import time
import uuid
import logging
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional

import requests

try:
    import openai
except ImportError:  # Allow tests / offline runs
    openai = None  # type: ignore

from synergesis.agents.core_agents import BaseAgent, AgentContext

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Configuration dataclasses
# ---------------------------------------------------------------------------


@dataclass
class NousConfig:
    base_url: str = os.getenv("NOUS_API_URL", "http://localhost:8001")
    timeout: int = 10  # seconds


@dataclass
class LLMConfig:
    provider: str = "openai"
    model: str = os.getenv("VYRA_MODEL", "gpt-4o-mini")
    api_key: Optional[str] = os.getenv("OPENAI_API_KEY")
    temperature: float = 0.7
    max_tokens: int = 256


# ---------------------------------------------------------------------------
# External service helpers
# ---------------------------------------------------------------------------


class NousClient:
    """Mini‑client REST pour interagir avec l’API NOUS."""

    def __init__(self, cfg: NousConfig):
        self.cfg = cfg

    # --- CRUD -------------------------------------------------------------

    def add_concept(self, label: str, data: Dict[str, Any]) -> Dict[str, Any]:
        payload = {"label": label, "data": data}
        r = requests.post(f"{self.cfg.base_url}/concepts", json=payload, timeout=self.cfg.timeout)
        r.raise_for_status()
        return r.json()

    def get_concept(self, concept_id: str) -> Dict[str, Any]:
        r = requests.get(f"{self.cfg.base_url}/concepts/{concept_id}", timeout=self.cfg.timeout)
        r.raise_for_status()
        return r.json()

    def search_concepts(self, query: str) -> List[Dict[str, Any]]:
        r = requests.get(f"{self.cfg.base_url}/concepts", params={"q": query}, timeout=self.cfg.timeout)
        r.raise_for_status()
        return r.json()


# ---------------------------------------------------------------------------
# LLM backend
# ---------------------------------------------------------------------------


class LLMBackend:
    """Interface simplifiée pour appeler un LLM."""

    def __init__(self, cfg: LLMConfig):
        self.cfg = cfg
        if self.cfg.provider == "openai" and openai:
            openai.api_key = self.cfg.api_key
            openai.base_url = os.getenv("OPENAI_BASE_URL", "https://api.openai.com/v1")

    def generate_concept(self, prompt: str) -> str:
        """Retourne un titre/label conceptuel d’après `prompt`."""
        if self.cfg.provider == "openai" and openai and self.cfg.api_key:
            response = openai.ChatCompletion.create(
                model=self.cfg.model,
                temperature=self.cfg.temperature,
                max_tokens=self.cfg.max_tokens,
                messages=[
                    {"role": "system", "content": "You are Vyra, an agent that suggests concise concept labels (≤7 words) based on input context."},
                    {"role": "user", "content": prompt},
                ],
            )
            return response.choices[0].message.content.strip()
        # --- Fallback stub
        logger.warning("LLM not available, returning stub concept.")
        return f"Stub‑Concept‑{uuid.uuid4().hex[:6]}"


# ---------------------------------------------------------------------------
# Vyra Agent
# ---------------------------------------------------------------------------


class Vyra(BaseAgent):
    """Agent créatif qui propose de nouveaux concepts et les inscrit dans NOUS."""

    def __init__(self, ctx: AgentContext, nous_cfg: Optional[NousConfig] = None, llm_cfg: Optional[LLMConfig] = None):
        super().__init__(ctx)
        self.nous = NousClient(nous_cfg or NousConfig())
        self.llm = LLMBackend(llm_cfg or LLMConfig())
        self.inbox: List[Dict[str, Any]] = []  # messages/glyphs en attente
        logger.info("Vyra initialisée.")

    # ------------------------------------------------------------------
    # API publique pour le bus
    # ------------------------------------------------------------------

    def receive(self, message: Dict[str, Any]):
        """Reçoit un message/glyph de la part du bus ou d’un autre agent."""
        self.inbox.append(message)

    # ------------------------------------------------------------------
    # Cycle de vie
    # ------------------------------------------------------------------

    def tick(self):
        """Traitement cyclique (appelé par agent_loop)."""
        if not self.inbox:
            return  # rien à faire

        glyph = self.inbox.pop(0)
        logger.debug("Vyra traite glyph %s", glyph.get("id"))
        concept_label = self._deduce_concept_label(glyph)
        if not concept_label:
            logger.debug("Aucun concept pertinent trouvé pour glyph %s", glyph.get("id"))
            return

        # Vérifier présence dans NOUS
        existing = self.nous.search_concepts(concept_label)
        if existing:
            logger.info("Concept déjà présent : '%s' (#%s)", concept_label, existing[0]["id"])
            return  # Tout est déjà dans NOUS

        # Générer description via LLM si label trop succinct
        enriched_label = self.llm.generate_concept(f"Create a refined concept name for: {concept_label}")
        data = {
            "source": "vyra",
            "derived_from": glyph.get("id"),
            "timestamp": time.time(),
            "confidence": glyph.get("confidence", 0.7),
        }
        new_concept = self.nous.add_concept(enriched_label, data)
        logger.info("Nouveau concept enregistré dans NOUS : %s (#%s)", new_concept["label"], new_concept["id"])

    # ------------------------------------------------------------------
    # Helpers internes
    # ------------------------------------------------------------------

    @staticmethod
    def _deduce_concept_label(glyph: Dict[str, Any]) -> Optional[str]:
        """Heuristique simple pour extraire un concept à partir d’un glyph."""
        tags = glyph.get("tags")
        if tags and isinstance(tags, list):
            # Choisit le tag le plus rare / long comme concept initial
            tags_sorted = sorted(tags, key=len, reverse=True)
            return tags_sorted[0].replace("_", " ").title()
        # Fallback : utiliser polarité+alignment
        pol = glyph.get("polarité", "?")
        aln = glyph.get("alignement", "Void")
        if isinstance(pol, str) and isinstance(aln, str):
            return f"{aln} {pol} Phenomenon".title()
        return None


# ---------------------------------------------------------------------------
# if run standalone (debug / simple test)
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)

    ctx = AgentContext(config={}, shared_state={})
    vyra = Vyra(ctx)

    # Glyph mock
    mock_glyph = {
        "id": "glyph‑demo‑001",
        "tags": ["quantum_flux", "ladderfall"],
        "polarité": "+",
        "alignement": "Celestial",
    }

    vyra.receive(mock_glyph)
    vyra.tick()
