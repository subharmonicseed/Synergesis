            with open(filename, 'a') as f: json.dump(glyph_data, f); f.write('\n'); logger_loop.info(f"Enqueued glyph {glyph_data['id']} from {source}.")
        except Exception as e: logger_loop.error(f"Failed to enqueue glyph {glyph_data['id']}: {e}")

    # --- Core Loop Tasks (called by scheduler or main loop) ---
    def task_process_new_glyphs(self):
        logger_loop.debug("Task: Processing new glyph files...")
        processed_count = 0
        try:
            if not os.path.exists(self.input_dir): os.makedirs(self.input_dir)
            if not os.path.exists(self.processed_dir): os.makedirs(self.processed_dir)
            for filename in os.listdir(self.input_dir):
                if filename.endswith((".jsonl", ".json")):
                    filepath = os.path.join(self.input_dir, filename)
                    logger_loop.info(f"Processing file: {filename}")
                    processed_in_file = 0
                    try:
                        with open(filepath, 'r') as f:
                            for line in f:
                                try: glyph_data = json.loads(line.strip());
                                if isinstance(glyph_data, dict) and 'id' in glyph_data: self._add_glyph_to_db(glyph_data); processed_in_file+=1
                                except json.JSONDecodeError: logger_loop.warning(f"Invalid JSON line in {filename}: {line.strip()[:50]}...")
                                except Exception as line_e: logger_loop.error(f"Error processing line in {filename}: {line_e}")
                        # Move file after successful processing of all lines
                        ts = datetime.now(timezone.utc).strftime('%Y%m%d%H%M%S')
                        processed_path = os.path.join(self.processed_dir, f"{filename}.processed_{ts}")
                        os.rename(filepath, processed_path); logger_loop.info(f"Moved {filename} to processed.")
                        processed_count += processed_in_file
                    except Exception as file_e: logger_loop.error(f"Error handling file {filename}: {file_e}", exc_info=True) # Keep file in input on error?
            # if processed_count > 0: logger_loop.info(f"Processed {processed_count} new glyph entries.")
        except Exception as e: logger_loop.error(f"Error checking input dir: {e}", exc_info=True)

    def task_translate_pending(self, limit: int = 10):
        logger_loop.debug(f"Task: Translating pending glyphs (limit {limit})...")
        conn = self._connect_db(); cursor = conn.cursor(); translated_count = 0
        try:
            cursor.execute("SELECT * FROM glyph_archive WHERE status = 'received' LIMIT ?", (limit,))
            rows = cursor.fetchall(); columns = [d[0] for d in cursor.description]
            for row in rows:
                glyph_dict = dict(zip(columns, row))
                try: glyph_dict['tags'] = json.loads(glyph_dict.get('tags', '[]'))
                except: glyph_dict['tags'] = []
                try: glyph_dict['sourceIds'] = json.loads(glyph_dict.get('sourceIds', '[]'))
                except: glyph_dict['sourceIds'] = []
                logger_loop.info(f"Translating glyph {glyph_dict['id']}...")
                translation = translate_glyph(glyph_dict)
                if "error" in translation: logger_loop.error(f"Translation failed for {glyph_dict['id']}: {translation['error']}")
                else:
                     # Add archetype during translation update
                     metrics = translation.get('metrics', {})
                     archetype = _cluster_signature(metrics.get('resonance',0), metrics.get('emergence',0), glyph_dict.get('alignement','?'), glyph_dict.get('polarité','?'), metrics.get('stability','?'))
                     translation['_archetype'] = archetype # Add to dict for update function
                     self._update_glyph_translation(glyph_dict['id'], translation) # Pass conn? No, update func handles conn
                     translated_count += 1
                time.sleep(0.05) # Tiny sleep between translations
        except Exception as e: logger_loop.error(f"Error during translation task: {e}", exc_info=True)
        finally: self._close_db(conn)
        # if translated_count > 0: logger_loop.info(f"Translation task complete. Processed: {translated_count}")

    def task_run_echo_check(self):
        logger_loop.info("Task: Running Echo Check...")
        try:
            report = self.echo_detector.detect_echoes()
            feedback_action = self.feedback_engine.process_stimulus({"type": "echo_report", "data": report})
            if feedback_action and feedback_action.get('action') == 'generate_glyph':
                self.enqueue_glyph(feedback_action['glyph_data'], "FeedbackEngine_Echo")
        except Exception as e: logger_loop.error(f"Error during Echo Check Task: {e}", exc_info=True)

    def task_run_llm_cycle(self):
        logger_loop.info("Task: Running LLM Prompt Cycle...")
        try:
            response_glyph = self.llm_generator.run_cycle()
            if response_glyph:
                self.enqueue_glyph(response_glyph, f"LLMResponse_{response_glyph.get('sourceIds',[''])[0].split(':')[-1]}") # Include source context in enqueue name
        except Exception as e: logger_loop.error(f"Error during LLM Cycle Task: {e}", exc_info=True)

    def task_run_observer(self):
        logger_loop.info("Task: Running Observer...")
        try:
            observation_glyph = self.observer.run_observation()
            if observation_glyph:
                self.enqueue_glyph(observation_glyph, "ObserverMode")
        except Exception as e: logger_loop.error(f"Error during Observer Task: {e}", exc_info=True)

    # --- Schedule Setup & Main Loop ---
    def setup_schedule(self):
        logger_loop.info("Setting up ritual schedule...")
        schedule.clear()
        schedule_data = load_config(DEFAULT_RITUAL_PATH, {}) # Load schedule file
        if not schedule_data or 'schedule' not in schedule_data: logger_loop.warning("No valid schedule found."); return

        tasks = { # Map task names in YAML to methods
            "ingest_check": self.task_process_new_glyphs, # Checks input dir
            "translate_batch": self.task_translate_pending,
            "echo_check": self.task_run_echo_check,
            "llm_prompt": self.task_run_llm_cycle,
            "observe": self.task_run_observer,
            # "coherence_check": self.task_run_coherence_check, # Add method if CoherenceTracker implemented
        }
        for job_name, details in schedule_data['schedule'].items():
            task_func = tasks.get(job_name)
            if not task_func: logger_loop.warning(f"Unknown job '{job_name}' in schedule."); continue
            try:
                 interval = int(details.get('interval', 5))
                 unit = details.get('unit', 'minutes').lower()
                 logger_loop.info(f"Scheduling '{job_name}' every {interval} {unit}...")
                 scheduler = schedule.every(interval)
                 job_instance = None
                 if unit == 'seconds': job_instance = scheduler.seconds
                 elif unit == 'minutes': job_instance = scheduler.minutes
                 elif unit == 'hours': job_instance = scheduler.hours
                 elif unit == 'days': job_instance = scheduler.days.at(details.get("at", "03:00")) # Daily specific time
                 elif unit == 'weeks': # Schedule library needs specific day
                      day = details.get("day", "sunday").lower(); at_time = details.get("at", "04:00")
                      if hasattr(schedule.every(), day): job_instance = getattr(schedule.every(), day).at(at_time)
                      else: logger_loop.warning(f"Invalid day '{day}' for weekly job '{job_name}'")
                 else: logger_loop.warning(f"Unsupported unit '{unit}' for job '{job_name}'")

                 if job_instance:
                      # Pass parameters if defined in schedule (e.g., limit for translate)
                      task_params = details.get('params', {})
                      job_instance.do(task_func, **task_params)
            except Exception as sched_e:
                 logger_loop.error(f"Error scheduling job '{job_name}': {sched_e}")


    def run_loop(self):
        """Runs the main orchestration loop indefinitely."""
        self.setup_schedule()
        logger_loop.info("Starting Agent Loop...")
        self.is_running = True
        while self.is_running:
            loop_start = time.monotonic()
            # Always run ingest check outside of schedule? Or rely purely on schedule?
            # Let's run ingest check frequently regardless of schedule
            self.task_process_new_glyphs()

            # Run scheduled tasks
            try:
                schedule.run_pending()
            except Exception as e:
                logger_loop.error(f"Error during schedule.run_pending(): {e}", exc_info=True)

            loop_duration = time.monotonic() - loop_start
            # Calculate sleep time intelligently based on next scheduled job
            idle_seconds = schedule.idle_seconds()
            if idle_seconds is None or idle_seconds <= 0:
                sleep_time = self.loop_sleep # Default sleep if no jobs or next is immediate
            else:
                sleep_time = max(0.5, min(self.loop_sleep, idle_seconds)) # Sleep up to default OR until next job

            # logger_loop.debug(f"Loop cycle duration: {loop_duration:.3f}s. Sleeping for {sleep_time:.2f}s.")
            time.sleep(sleep_time)
