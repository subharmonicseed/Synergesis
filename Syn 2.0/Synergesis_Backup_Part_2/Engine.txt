// glyphUtils.js
/**
 * @file Utility functions and constants for the Syn Symbolic Glyph Tool.
 * Contains shared constants, random generation helpers, geometric calculations,
 * object manipulation, and drawing primitives.
 */

// --- Symbolic Constants & Configuration ---

/**
 * Defines the core symbolic properties and their possible values or ranges.
 * @const {object} SYMBOLIC_PROPERTIES
 * @property {string[]} POLARITY - Possible polarity states: Positive, Negative, Neutral.
 * @property {string[]} ALIGNMENT - Possible symbolic alignments (examples).
 * @property {number[]} BASE_FREQUENCY_RANGE - Min/Max base frequency for glyphs.
 * @property {number[]} BASE_WEIGHT_RANGE - Min/Max base energetic weight for glyphs.
 * @property {number} COMPATIBILITY_THRESHOLD_HIGH - Normalized score above which compatibility is 'compatible'.
 * @property {number} COMPATIBILITY_THRESHOLD_LOW - Normalized score below which compatibility is 'incompatible'.
 */
const SYMBOLIC_PROPERTIES = {
    POLARITY: ['+', '-', '0'],
    ALIGNMENT: ['celestial', 'terrestrial', 'chthonic', 'void', 'elemental', 'harmonic', 'discordant'], // Expanded list
    BASE_FREQUENCY_RANGE: [1, 100],
    BASE_WEIGHT_RANGE: [1, 10],
    COMPATIBILITY_THRESHOLD_HIGH: 0.7, // Increased slightly
    COMPATIBILITY_THRESHOLD_LOW: 0.35, // Increased slightly
};

/**
 * Maps geometric shape types to their inherent symbolic tendencies (hints/biases).
 * Used during glyph generation to influence the final calculated properties.
 * @const {object.<string, object>} SHAPE_SYMBOL_MAP
 * @property {object} [shapeType] - Configuration for a specific shape type.
 * @property {string|string[]} [shapeType].polarityBias - Tendency towards a polarity ('+', '-', '0', or array for random choice).
 * @property {number} [shapeType].freqMod - Multiplier affecting frequency calculation.
 * @property {number} [shapeType].weightMod - Base contribution to the glyph's weight.
 * @property {string[]} [shapeType].alignmentHint - Possible alignments associated with this shape.
 * @property {string[]} [shapeType].tags - Specific tags added to the glyph if this shape is present.
 */
const SHAPE_SYMBOL_MAP = {
    circle: { polarityBias: '0', freqMod: 1.0, weightMod: 1.2, alignmentHint: ['celestial', 'harmonic'], tags: ['continuity', 'unity'] },
    triangle: { polarityBias: ['+', '-'], freqMod: 1.1, weightMod: 1.0, alignmentHint: ['elemental', 'chthonic', 'discordant'], tags: ['focus', 'direction', 'tension'] },
    line: { polarityBias: '0', freqMod: 0.9, weightMod: 0.8, alignmentHint: ['terrestrial', 'void'], tags: ['connection', 'separation', 'path'] },
    arc: { polarityBias: '0', freqMod: 1.05, weightMod: 1.1, alignmentHint: ['celestial', 'elemental', 'harmonic'], tags: ['flow', 'opening', 'cycle'] },
    // Add more complex shapes maybe later? e.g., rectangle, cross
};

// --- General Utility Functions ---

/**
 * Generates a random floating-point number within a specified range.
 * @param {number} min - The minimum value (inclusive).
 * @param {number} max - The maximum value (exclusive).
 * @returns {number} A random number between min and max.
 */
function randomRange(min, max) {
    return Math.random() * (max - min) + min;
}

/**
 * Generates a random integer within a specified range (inclusive).
 * @param {number} min - The minimum integer value.
 * @param {number} max - The maximum integer value.
 * @returns {number} A random integer between min and max.
 */
function randomInt(min, max) {
    min = Math.ceil(min);
    max = Math.floor(max);
    return Math.floor(Math.random() * (max - min + 1)) + min;
}

/**
 * Selects a random element from an array.
 * @param {Array<*>} arr - The array to choose from.
 * @returns {*} A random element from the array, or undefined if the array is empty.
 */
function randomChoice(arr) {
    if (!arr || arr.length === 0) return undefined;
    return arr[Math.floor(Math.random() * arr.length)];
}

/**
 * Generates a simple timestamp-based unique ID. Sufficient for internal use.
 * Not cryptographically secure or guaranteed globally unique like UUID v4.
 * @returns {string} A unique identifier string.
 */
function generateUUID() {
    // Simple implementation for internal tool uniqueness
    return `syn-${Date.now().toString(36)}-${Math.random().toString(36).substring(2, 9)}`;
}

/**
 * Performs a deep clone of an object using JSON stringify/parse.
 * Works for simple objects, arrays, and primitives. Loses functions, Dates become strings, etc.
 * Sufficient for cloning glyph data for archiving.
 * @param {object} obj - The object to clone.
 * @returns {object|null} The cloned object, or null if cloning fails.
 */
function deepClone(obj) {
    if (obj === null || typeof obj !== 'object') {
        return obj; // Return primitives directly
    }
    try {
        // Handle potential Date objects (convert back after parse)
        // This basic version doesn't handle Dates, Functions, Regexps, Maps, Sets correctly.
        // For this tool's data structure (simple objects, arrays, strings, numbers), it's okay.
        return JSON.parse(JSON.stringify(obj));
    } catch (e) {
        console.error("DeepClone Error:", e, "Object:", obj);
        return null; // Indicate failure
    }
}

/**
 * Normalizes a value to a range between 0 and 1.
 * @param {number} value - The value to normalize.
 * @param {number} min - The minimum possible value in the original range.
 * @param {number} max - The maximum possible value in the original range.
 * @returns {number} The normalized value (clamped between 0 and 1).
 */
function normalize(value, min, max) {
    if (max === min) return 0.5; // Avoid division by zero, return neutral value
    return Math.max(0, Math.min(1, (value - min) / (max - min)));
}

/**
 * Calculates the mouse position relative to the canvas element,
 * accounting for CSS scaling and border/padding.
 * @param {HTMLCanvasElement} canvas - The canvas element.
 * @param {MouseEvent} event - The mouse event.
 * @returns {{x: number, y: number}} The coordinates {x, y} relative to the canvas's internal resolution.
 */
function getCanvasMousePos(canvas, event) {
     const rect = canvas.getBoundingClientRect();
     // Calculate scale based on actual rendered size vs internal resolution
     const scaleX = canvas.width / rect.width;
     const scaleY = canvas.height / rect.height;
     // Calculate mouse position relative to the canvas top-left
     const canvasX = (event.clientX - rect.left) * scaleX;
     const canvasY = (event.clientY - rect.top) * scaleY;
     return { x: canvasX, y: canvasY };
}

/**
 * Draws a single geometric shape on the canvas context.
 * Assumes context transformations (translate) are handled externally for glyph positioning.
 * Positions within the shape object are relative to the glyph's center (0,0 after translate).
 * @param {CanvasRenderingContext2D} ctx - The canvas rendering context.
 * @param {object} shape - The shape object containing type and geometric properties.
 * @param {string} shape.type - Type of shape ('circle', 'triangle', 'line', 'arc').
 * @param {number} shape.x - Relative X position.
 * @param {number} shape.y - Relative Y position.
 * @param {number} [shape.radius] - Radius (for circle, arc).
 * @param {number} [shape.size] - Size parameter (e.g., base for triangle).
 * @param {number} [shape.angle] - Rotation angle in radians (for triangle, line).
 * @param {number} [shape.length] - Length (for line).
 * @param {number} [shape.startAngle] - Start angle in radians (for arc).
 * @param {number} [shape.endAngle] - End angle in radians (for arc).
 */
function drawShape(ctx, shape) {
    ctx.beginPath();
    const x = shape.x; // Positions are already relative
    const y = shape.y;

    switch (shape.type) {
        case 'circle':
            if (typeof shape.radius !== 'number') { console.warn("Missing radius for circle:", shape); return; }
            ctx.arc(x, y, shape.radius, 0, Math.PI * 2);
            break;
        case 'triangle':
             if (typeof shape.size !== 'number' || typeof shape.angle !== 'number') { console.warn("Missing size/angle for triangle:", shape); return; }
             // Equilateral triangle for simplicity
            ctx.save();
            ctx.translate(x, y);
            ctx.rotate(shape.angle);
            const h = shape.size * (Math.sqrt(3) / 2); // Height
            ctx.moveTo(0, -h / 2); // Top point
            ctx.lineTo(-shape.size / 2, h / 2); // Bottom left
            ctx.lineTo(shape.size / 2, h / 2); // Bottom right
            ctx.closePath();
            ctx.restore();
            break;
        case 'line':
             if (typeof shape.length !== 'number' || typeof shape.angle !== 'number') { console.warn("Missing length/angle for line:", shape); return; }
             // Draw line centered around (x, y) with given length and angle
             const dx = Math.cos(shape.angle) * shape.length / 2;
             const dy = Math.sin(shape.angle) * shape.length / 2;
             ctx.moveTo(x - dx, y - dy);
             ctx.lineTo(x + dx, y + dy);
            break;
        case 'arc':
            if (typeof shape.radius !== 'number' || typeof shape.startAngle !== 'number' || typeof shape.endAngle !== 'number') { console.warn("Missing radius/angles for arc:", shape); return; }
            ctx.arc(x, y, shape.radius, shape.startAngle, shape.endAngle);
            break;
        default:
            console.warn(`Attempted to draw unknown shape type: ${shape.type}`, shape);
            return; // Don't stroke unknown shapes
    }
    ctx.stroke(); // Apply the stroke defined before calling this function
}

// --- Text Wrapping for Analysis Panel ---
/**
 * Simple text wrapping function for drawing text within a constrained width.
 * Used primarily if drawing analysis directly onto canvas was needed (not currently used for DOM panel).
 * @param {CanvasRenderingContext2D} ctx - Canvas context.
 * @param {string} text - Text to wrap.
 * @param {number} x - Starting X position.
 * @param {number} y - Starting Y position.
 * @param {number} maxWidth - Maximum width for a line.
 * @param {number} lineHeight - Height of each line.
 */
function wrapText(ctx, text, x, y, maxWidth, lineHeight) {
    const words = text.split(' ');
    let line = '';
    for(let n = 0; n < words.length; n++) {
      const testLine = line + words[n] + ' ';
      const metrics = ctx.measureText(testLine);
      const testWidth = metrics.width;
      if (testWidth > maxWidth && n > 0) {
        ctx.fillText(line, x, y);
        line = words[n] + ' ';
        y += lineHeight;
      } else {
        line = testLine;
      }
    }
    ctx.fillText(line, x, y);
}

        /**
         * Creates a Glyph instance.
         * @param {number} x - Center X coordinate on the canvas.
         * @param {number} y - Center Y coordinate on the canvas.
         * @param {string} [id=generateUUID()] - Unique identifier. If null, generated automatically.
         * @param {object[]} [shapes=null] - Array of shape objects. If null, generated automatically.
         * @param {object} [properties=null] - Pre-calculated symbolic properties. If null, calculated from shapes.

            this.id = id || generateUUID();

             * @type {object[]}

             * @property {number} x - Relative X position.
             * @property {number} y - Relative Y position.

             * @property {object} symbolicHints - Hints derived from SHAPE_SYMBOL_MAP.

             * Calculated symbolic properties of the glyph.
             * @type {object}
             * @property {string} polarity - Dominant polarity ('+', '-', '0').
             * @property {number} frequency - Calculated frequency (integer).
             * @property {number} weight - Calculated energetic weight (integer).
             * @property {string} alignment - Dominant symbolic alignment.
             * @property {string[]} values - Array of associated tags/keywords.

        /**
         * Generates the geometric shapes for this glyph based on configured parameters.
         * Each shape gets symbolic hints assigned based on SHAPE_SYMBOL_MAP.
         * @returns {object[]} An array of shape objects.
         */
        generateShapes() {
            const shapes = [];
            const numShapes = randomInt(MIN_SHAPES_PER_GLYPH, MAX_SHAPES_PER_GLYPH);
            const availableShapeTypes = Object.keys(SHAPE_SYMBOL_MAP);

            for (let i = 0; i < numShapes; i++) {
                const type = randomChoice(availableShapeTypes);
                if (!type) continue; // Should not happen if SHAPE_SYMBOL_MAP is populated

                const shape = { type };
                const shapeSymbolData = SHAPE_SYMBOL_MAP[type];

                // Position relative to glyph center (this.x, this.y will be applied during draw)
                shape.x = randomRange(-BASE_GLYPH_SIZE * 0.38, BASE_GLYPH_SIZE * 0.38);
                shape.y = randomRange(-BASE_GLYPH_SIZE * 0.38, BASE_GLYPH_SIZE * 0.38);
                const maxSize = BASE_GLYPH_SIZE * SHAPE_MAX_SIZE_FACTOR;
                const size = randomRange(maxSize * 0.15, maxSize); // Ensure some variation
                shape.angle = randomRange(0, Math.PI * 2); // Generic angle

                // Assign specific geometric properties based on type
                switch (type) {
                    case 'circle': shape.radius = size / 2; break;
                    case 'triangle': shape.size = size; break; // Base width
                    case 'line': shape.length = size; break;
                    case 'arc':
                        shape.radius = size / 2;
                        shape.startAngle = shape.angle; // Use generic angle as start
                        shape.endAngle = shape.angle + randomRange(Math.PI * 0.4, Math.PI * 1.9); // Significant arc length
                        break;
                }

                // Assign symbolic hints from the map
                shape.symbolicHints = {
                    polarityBias: Array.isArray(shapeSymbolData.polarityBias) ? randomChoice(shapeSymbolData.polarityBias) : shapeSymbolData.polarityBias,
                    freqMod: shapeSymbolData.freqMod || 1.0,
                    weightMod: shapeSymbolData.weightMod || 1.0,
                    alignmentHint: randomChoice(shapeSymbolData.alignmentHint || SYMBOLIC_PROPERTIES.ALIGNMENT), // Fallback to any alignment
                    tags: shapeSymbolData.tags ? [...shapeSymbolData.tags] : [] // Copy tags array
                };
                shapes.push(shape);
            }
            return shapes;
        }

        /**
         * Calculates the aggregated symbolic properties of the glyph based on its shapes.
         * This is the core symbolic logic for generation.
         * @returns {object} The calculated properties object.
         */
        calculateSymbolicProperties() {
            let totalWeightContribution = 0;
            let weightedFreqSum = 0;
            let polarityVotes = { '+': 0, '-': 0, '0': 0 };
            let alignmentVotes = {};
            SYMBOLIC_PROPERTIES.ALIGNMENT.forEach(a => alignmentVotes[a] = 0);
            let allTags = new Set();

            this.shapes.forEach(shape => {
                const hints = shape.symbolicHints;
                const shapeWeight = hints.weightMod;

                totalWeightContribution += shapeWeight;
                // Base frequency influenced by modifier and weighted by shape's contribution
                weightedFreqSum += randomRange(...SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE) * hints.freqMod * shapeWeight;

                // Tally votes for polarity and alignment
                if (polarityVotes[hints.polarityBias] !== undefined) {
                    polarityVotes[hints.polarityBias]++;
                }
                if (alignmentVotes[hints.alignmentHint] !== undefined) {
                     alignmentVotes[hints.alignmentHint]++;
                } else {
                    // Fallback if hint is somehow invalid
                    alignmentVotes[randomChoice(SYMBOLIC_PROPERTIES.ALIGNMENT)]++;
                }

                // Collect all unique tags
                hints.tags.forEach(tag => allTags.add(tag));
                allTags.add(`shape:${shape.type}`); // Add shape type as a tag
            });

            // Determine dominant polarity
            let dominantPolarity = '0'; // Default to neutral
            const posVotes = polarityVotes['+'];
            const negVotes = polarityVotes['-'];
            const neuVotes = polarityVotes['0'];
            if (posVotes > negVotes && posVotes >= neuVotes) dominantPolarity = '+';
            else if (negVotes > posVotes && negVotes >= neuVotes) dominantPolarity = '-';
            // If pos == neg and they are > neu, or if neu is highest, remains '0'

             // Determine dominant alignment
            let dominantAlignment = randomChoice(SYMBOLIC_PROPERTIES.ALIGNMENT); // Default random
            let maxVotes = -1;
             let contenders = [];
             for (const align in alignmentVotes) {
                 if (alignmentVotes[align] > maxVotes) {
                     maxVotes = alignmentVotes[align];
                     contenders = [align]; // New leader
                 } else if (alignmentVotes[align] === maxVotes) {
                     contenders.push(align); // Tie
                 }
             }
             if (contenders.length > 0) {
                 dominantAlignment = randomChoice(contenders); // Choose randomly among tied leaders
             }


            // Calculate final frequency and weight, clamped to ranges
            const finalFrequency = totalWeightContribution > 0
                ? Math.round(weightedFreqSum / totalWeightContribution)
                : randomInt(...SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE); // Avoid division by zero

            const finalWeight = totalWeightContribution > 0
                ? totalWeightContribution * (SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[1] / (MAX_SHAPES_PER_GLYPH * 1.1)) // Rough scaling based on max potential weight
                : randomInt(...SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE);

            return {
                polarity: dominantPolarity,
                frequency: Math.max(SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[0], Math.min(SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[1], finalFrequency)),
                weight: Math.round(Math.max(SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[0], Math.min(SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[1], finalWeight))),
                alignment: dominantAlignment,
                values: Array.from(allTags).sort(), // Ensure consistent order
                complexity: this.shapes.length,
                sourceIds: [] // Empty for newly generated glyphs
            };
        }

         * Handles applying transformations and drawing selection highlight.
         * @param {CanvasRenderingContext2D} context - The rendering context.

        draw(context) {
            context.save();
            context.translate(this.x, this.y); // Move origin to glyph center

            context.lineWidth = RENDER_STROKE_WIDTH;
            context.strokeStyle = RENDER_STROKE_COLOR;

                drawShape(context, shape); // Use the utility function

                context.beginPath();
                // Draw a circle slightly larger than the interaction radius
                context.arc(0, 0, this.interactionRadius + 4, 0, Math.PI * 2);
                context.stroke();
            }

            context.restore(); // Restore context state
        }

        /**
         * Checks if a given point (px, py) is within the glyph's interaction radius.
         * @param {number} px - The X coordinate of the point to check.

        contains(px, py) {
             // Simple circular collision detection

        }

        toJSON() {

    /**
     * Handles click events on the canvas for glyph selection/deselection.
     * @param {MouseEvent} event - The click event object.

    function handleCanvasClick(event) {
        const { x, y } = getCanvasMousePos(canvas, event);
        const clickedGlyph = findGlyphAt(x, y);

        if (event.shiftKey) { // Shift-click toggles selection without limit (useful for API testing maybe?)
             if (clickedGlyph) {
                 clickedGlyph.isSelected = !clickedGlyph.isSelected;
                 if (clickedGlyph.isSelected) {
                     if (!selectedGlyphs.includes(clickedGlyph)) selectedGlyphs.push(clickedGlyph);
                 } else {
                     selectedGlyphs = selectedGlyphs.filter(g => g !== clickedGlyph);
                 }
             }
        } else { // Normal click logic (max 2 selections)
            if (clickedGlyph) {
                if (selectedGlyphs.includes(clickedGlyph)) {
                    // Clicked on already selected: Deselect it
                    clickedGlyph.isSelected = false;
                    selectedGlyphs = selectedGlyphs.filter(g => g !== clickedGlyph);
                } else {
                    // Clicked on unselected glyph
                    if (selectedGlyphs.length < 2) {
                        // Less than 2 selected: Add to selection
                        clickedGlyph.isSelected = true;
                        selectedGlyphs.push(clickedGlyph);
                    } else {
                        // Exactly 2 selected: Replace the first selected with the new one
                        const firstSelected = selectedGlyphs.shift(); // Remove first
                        if(firstSelected) firstSelected.isSelected = false; // Deselect visually
                        clickedGlyph.isSelected = true; // Select the new one
                        selectedGlyphs.push(clickedGlyph); // Add new one at the end
                    }
                }
            } else {
                // Clicked outside any glyph: Deselect all
                selectedGlyphs.forEach(g => g.isSelected = false);
                selectedGlyphs = [];
            }
        }

        // Clear archive selection when interacting with canvas
        if(selectedArchiveId) {
            selectedArchiveId = null;
            // Need to redraw archive to remove highlight
            displayArchive(searchArchiveInput.value);
        }

        updateUIState(); // Update buttons and analysis panel
        requestAnimationFrame(drawAllGlyphs); // Redraw canvas

    /**
     * Tries to find a suitable position on the canvas for a new glyph, avoiding overlaps.
     * @returns {{x: number, y: number}|null} Coordinates {x, y} or null if no space found.
     */
    function findPlacementPosition() {
        let x, y, attempts = 0, overlaps;
         // Use effective canvas size (style dimensions) for placement logic
        const effectiveWidth = parseFloat(canvas.style.width);
        const effectiveHeight = parseFloat(canvas.style.height);
        const radius = BASE_GLYPH_SIZE / 2; // Use logical size for placement

        do {
            overlaps = false;
            // Calculate valid range considering glyph radius and padding
            const minX = radius + CANVAS_PADDING;
            const maxX = effectiveWidth - radius - CANVAS_PADDING;
            const minY = radius + CANVAS_PADDING;
            const maxY = effectiveHeight - radius - CANVAS_PADDING;

             // Check if canvas is too small
             if (maxX <= minX || maxY <= minY) {
                 console.warn("Canvas too small for optimal placement.");
                 // Fallback: place near center, even if it overlaps
                 return { x: effectiveWidth / 2, y: effectiveHeight / 2 };
             }

            x = randomRange(minX, maxX);
            y = randomRange(minY, maxY);

            // Check for overlap with existing glyphs based on interaction radius
            for (const existingGlyph of currentGlyphs) {
                const distSq = (x - existingGlyph.x)**2 + (y - existingGlyph.y)**2;
                // If distance^2 < (radius1 + radius2)^2, they overlap
                if (distSq < (radius + existingGlyph.interactionRadius)**2) {
                    overlaps = true;
                    break;
                }
            }
            attempts++;
        } while (overlaps && attempts < MAX_PLACEMENT_ATTEMPTS);

        if (attempts >= MAX_PLACEMENT_ATTEMPTS) {
            console.warn(`Placement Warning: Could not find non-overlapping position after ${attempts} attempts. Placing anyway.`);
            // Still return the last attempted position
        }
        return { x, y };
    }

    /**
     * Generates a new glyph, either automatically or based on specified rules.
     * @param {string} mode - Generation mode: 'auto' or 'rules'.
     * @param {object} [rules={}] - Rules object (used only if mode is 'rules').
     * @property {string} [rules.polarity] - Target polarity ('+', '-', '0', 'any').
     * @property {number} [rules.freqMin] - Min target frequency.
     * @property {number} [rules.freqMax] - Max target frequency.
     * @property {number} [rules.weightMin] - Min target weight.
     * @property {number} [rules.weightMax] - Max target weight.
     * @returns {Glyph|null} The newly generated Glyph object or null if generation failed (e.g., rules too strict).
     */
    function generateNewGlyph(mode = 'auto', rules = {}) {
        const position = findPlacementPosition();
        if (!position) {
            console.error("Generation failed: Could not find placement position.");
            showTemporaryMessage("Erreur: Impossible de placer un nouveau glyphe.", true);
            return null;
        }

        let newGlyph = null;
        let attempts = 0;

        if (mode === 'auto') {
            newGlyph = new Glyph(position.x, position.y);
            currentGlyphs.push(newGlyph);
        } else if (mode === 'rules') {
            // Attempt to generate a glyph matching the rules
            do {
                const candidateGlyph = new Glyph(position.x, position.y);
                if (checkGlyphAgainstRules(candidateGlyph, rules)) {
                    newGlyph = candidateGlyph;
                    break; // Found a matching glyph
                }
                attempts++;
            } while (attempts < MAX_GENERATION_ATTEMPTS);

            if (newGlyph) {
                currentGlyphs.push(newGlyph);
                console.log(`Generated glyph matching rules after ${attempts + 1} attempt(s).`);
            } else {
                console.warn(`Generation failed: Could not create glyph matching rules after ${MAX_GENERATION_ATTEMPTS} attempts.`);
                 showTemporaryMessage("Échec: Règles trop strictes, impossible de générer.", true);
                return null; // Indicate failure
            }
        } else {
            console.error("Invalid generation mode specified:", mode);
            return null;
        }

        requestAnimationFrame(drawAllGlyphs);
        updateUIState(); // Update counts, etc.
        return newGlyph; // Return the generated glyph
    }

    /**
     * Checks if a generated glyph's properties match the specified rules.
     * @param {Glyph} glyph - The glyph to check.
     * @param {object} rules - The rules object (same structure as in generateNewGlyph).
     * @returns {boolean} True if the glyph matches the rules, false otherwise.
     */
    function checkGlyphAgainstRules(glyph, rules) {
        const props = glyph.properties;
        if (rules.polarity && rules.polarity !== 'any' && props.polarity !== rules.polarity) {
            return false;
        }
        if (typeof rules.freqMin === 'number' && props.frequency < rules.freqMin) {
            return false;
        }
        if (typeof rules.freqMax === 'number' && props.frequency > rules.freqMax) {
            return false;
        }
        if (typeof rules.weightMin === 'number' && props.weight < rules.weightMin) {
            return false;
        }
         if (typeof rules.weightMax === 'number' && props.weight > rules.weightMax) {
            return false;
        }
        // Add checks for other rules (e.g., alignment, specific tags) if implemented
        return true; // All checks passed
    }

    // --- Symbolic Analysis & Combination ---

    /**
     * Calculates the symbolic compatibility between two glyphs.
     * Provides a numerical score, a categorical level, and detailed reasons.
     * @param {Glyph} glyph1 - The first glyph.
     * @param {Glyph} glyph2 - The second glyph.
     * @returns {{score: number, normalized: number, level: string, reasons: string[]}} Compatibility details.
     */
    function calculateSymbolicCompatibility(glyph1, glyph2) {
        let score = 0;
        const reasons = [];
        const p1 = glyph1.properties;
        const p2 = glyph2.properties;

        // 1. Polarity Compatibility (+/- attract, 0 compatible, same sign repel)
        let polarityScore = 0;
        if ((p1.polarity === '+' && p2.polarity === '-') || (p1.polarity === '-' && p2.polarity === '+')) {
            polarityScore = 2.5; // Strong attraction
            reasons.push(`<strong>Polarité:</strong> Attraction forte (${p1.polarity}, ${p2.polarity}).`);
        } else if (p1.polarity === '0' || p2.polarity === '0') {
            polarityScore = 1.0; // Neutral is generally compatible
            reasons.push(`<strong>Polarité:</strong> Compatibilité neutre (${p1.polarity}, ${p2.polarity}).`);
        } else if (p1.polarity === p2.polarity) {
            polarityScore = -1.5; // Identical non-neutral signs repel
            reasons.push(`<strong>Polarité:</strong> Répulsion (${p1.polarity}, ${p2.polarity} identiques).`);
        }
        score += polarityScore;

        // 2. Frequency Resonance (Closer is better)
        const freqDiff = Math.abs(p1.frequency - p2.frequency);
        const maxFreqDiff = SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[1] - SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[0];
        const freqSimilarity = 1.0 - normalize(freqDiff, 0, maxFreqDiff); // 1 = identical, 0 = max diff
        const freqScore = freqSimilarity * 2.0; // Weight factor for frequency
        score += freqScore;
        reasons.push(`<strong>Fréquence:</strong> Résonance ${freqSimilarity > 0.8 ? 'élevée' : freqSimilarity > 0.5 ? 'modérée' : 'faible'} (Sim: ${freqSimilarity.toFixed(2)}, Freq: ${p1.frequency}, ${p2.frequency}).`);

        // 3. Alignment Harmony/Conflict
        let alignmentScore = 0;
        if (p1.alignment === p2.alignment) {
            alignmentScore = 1.5; // Identical alignment is harmonious
            reasons.push(`<strong>Alignement:</strong> Harmonie (${p1.alignment} identique).`);
        } else {
            // Define simple conflicts/synergies (example)
            const conflictPairs = [['celestial', 'chthonic'], ['harmonic', 'discordant']];
            const synergyPairs = [['elemental', 'terrestrial']]; // Example
            let isConflict = conflictPairs.some(pair => (pair.includes(p1.alignment) && pair.includes(p2.alignment)));
            let isSynergy = synergyPairs.some(pair => (pair.includes(p1.alignment) && pair.includes(p2.alignment)));

            if (isConflict) {
                alignmentScore = -1.0;
                reasons.push(`<strong>Alignement:</strong> Conflit potentiel (${p1.alignment} vs ${p2.alignment}).`);
            } else if (isSynergy) {
                alignmentScore = 0.75;
                reasons.push(`<strong>Alignement:</strong> Synergie potentielle (${p1.alignment} & ${p2.alignment}).`);
            } else {
                alignmentScore = -0.25; // Default slight incompatibility for difference
                reasons.push(`<strong>Alignement:</strong> Différents mais neutres (${p1.alignment}, ${p2.alignment}).`);
            }
        }
        score += alignmentScore;

        // 4. Energetic Weight Balance
        const weightDiff = Math.abs(p1.weight - p2.weight);
        const maxWeightDiff = SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[1] - SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[0];
        // Balance is higher when difference is lower
        const weightBalance = 1.0 - normalize(weightDiff, 0, maxWeightDiff * 0.8); // Penalize large differences more
        const weightScore = weightBalance * 1.0; // Weight factor for balance
        score += weightScore;
         reasons.push(`<strong>Poids:</strong> Équilibre ${weightBalance > 0.7 ? 'bon' : weightBalance > 0.4 ? 'moyen' : 'faible'} (Bal: ${weightBalance.toFixed(2)}, W: ${p1.weight}, ${p2.weight}).`);

         // 5. Complexity Similarity (Similar number of shapes might blend better)
         const complexityDiff = Math.abs(p1.complexity - p2.complexity);
         const maxComplexityDiff = MAX_SHAPES_PER_GLYPH - MIN_SHAPES_PER_GLYPH;
         const complexitySimilarity = 1.0 - normalize(complexityDiff, 0, maxComplexityDiff);
         const complexityScore = complexitySimilarity * 0.5; // Lower weight
         score += complexityScore;
         reasons.push(`<strong>Complexité:</strong> Similitude ${complexitySimilarity > 0.7 ? 'élevée' : 'modérée'} (Sim: ${complexitySimilarity.toFixed(2)}, C: ${p1.complexity}, ${p2.complexity}).`);


        // --- Final Score Calculation & Categorization ---
        // Estimate min/max possible scores based on weights above (approximate)
        const maxPossibleScore = 2.5 + 2.0 + 1.5 + 1.0 + 0.5; // ~7.5
        const minPossibleScore = -1.5 + 0 - 1.0 + 0 + 0; // ~ -2.5
        const normalizedScore = normalize(score, minPossibleScore, maxPossibleScore);

        let compatibilityLevel = 'neutral';
        if (normalizedScore >= SYMBOLIC_PROPERTIES.COMPATIBILITY_THRESHOLD_HIGH) {
            compatibilityLevel = 'compatible';
        } else if (normalizedScore <= SYMBOLIC_PROPERTIES.COMPATIBILITY_THRESHOLD_LOW) {
            compatibilityLevel = 'incompatible';
        }

        return {
            score: parseFloat(score.toFixed(2)),
            normalized: parseFloat(normalizedScore.toFixed(2)),
            level: compatibilityLevel,
            reasons: reasons
        };
    }

    /**
     * Merges two selected glyphs into a new, symbolically coherent glyph.
     * Implements the advanced fusion logic based on compatibility and properties.
     */
    function combineSelectedGlyphs() {
        if (selectedGlyphs.length !== 2) {
            console.warn("Combine function called without exactly 2 selected glyphs.");
            return; // Should be prevented by UI state, but double-check
        }

        const [glyph1, glyph2] = selectedGlyphs;
        const compatibility = calculateSymbolicCompatibility(glyph1, glyph2);
        console.log(`Attempting to combine ${glyph1.id} and ${glyph2.id}. Compatibility: ${compatibility.level} (Score: ${compatibility.score})`);

        // --- Fusion Logic ---

        // 1. Shape Fusion: Blend shapes, possibly add a 'binding' shape
        const shapes1 = deepClone(glyph1.shapes);
        const shapes2 = deepClone(glyph2.shapes);
        const newShapes = [];
        // Take more shapes from the more 'dominant' (e.g., heavier) glyph, or based on compatibility?
        // Simple approach: take roughly half from each, minimum 1.
        const numToTake1 = Math.max(1, Math.floor(shapes1.length * randomRange(0.4, 0.6)));
        const numToTake2 = Math.max(1, Math.floor(shapes2.length * randomRange(0.4, 0.6)));

        for (let i = 0; i < numToTake1 && shapes1.length > 0; i++) {
            newShapes.push(shapes1.splice(randomInt(0, shapes1.length - 1), 1)[0]);
        }
        for (let i = 0; i < numToTake2 && shapes2.length > 0; i++) {
            newShapes.push(shapes2.splice(randomInt(0, shapes2.length - 1), 1)[0]);
        }

         // Add a subtle binding element if compatibility is not 'incompatible'
         if (compatibility.level !== 'incompatible') {
             newShapes.push({
                 type: randomChoice(['line', 'arc']), // Line or Arc as binder
                 x: randomRange(-5, 5), y: randomRange(-5, 5), // Slightly offset
                 radius: BASE_GLYPH_SIZE * 0.1, size: BASE_GLYPH_SIZE * 0.15, length: BASE_GLYPH_SIZE * 0.15, // Small size
                 angle: randomRange(0, Math.PI * 2), startAngle: randomRange(0, Math.PI*2), endAngle: randomRange(0, Math.PI*2),
                 symbolicHints: { // Neutral binder
                     polarityBias: '0', freqMod: 1, weightMod: 0.3, // Very low weight
                     alignmentHint: 'harmonic', tags: ['binding', 'fusion']
                 }
             });
         }

         // Jitter positions slightly to avoid perfect overlap if shapes are identical
         newShapes.forEach(shape => {
             shape.x += randomRange(-3, 3);
             shape.y += randomRange(-3, 3);
         });

        // 2. Property Calculation for Fused Glyph
        const newProps = {};
        const p1 = glyph1.properties;
        const p2 = glyph2.properties;
        const totalWeightParents = Math.max(1, p1.weight + p2.weight); // Avoid division by zero

        // Polarity: Weighted influence, neutral if opposite unless highly unbalanced weight
        if (p1.polarity === p2.polarity) {
            newProps.polarity = p1.polarity;
        } else if ((p1.polarity === '+' && p2.polarity === '-') || (p1.polarity === '-' && p2.polarity === '+')) {
            const weightRatio = p1.weight / totalWeightParents;
            if (weightRatio > 0.7 || weightRatio < 0.3) { // If one is much heavier
                newProps.polarity = p1.weight > p2.weight ? p1.polarity : p2.polarity;
            } else { newProps.polarity = '0'; } // Balanced opposites -> neutral
        } else { // One is neutral
            newProps.polarity = p1.polarity !== '0' ? p1.polarity : p2.polarity;
        }

        // Frequency: Weighted average based on parent weights
        newProps.frequency = Math.round((p1.frequency * p1.weight + p2.frequency * p2.weight) / totalWeightParents);

        // Weight: Sum of parents, modified by compatibility level
        newProps.weight = p1.weight + p2.weight;
        if (compatibility.level === 'compatible') newProps.weight *= 1.15; // Synergy bonus
        else if (compatibility.level === 'incompatible') newProps.weight *= 0.85; // Conflict penalty
        newProps.weight = Math.round(newProps.weight);

        // Alignment: If same, keep. If different, weighted choice or specific 'mixed' state.
        if (p1.alignment === p2.alignment) {
            newProps.alignment = p1.alignment;
        } else {
             // Weighted random choice based on parent weights
             const choice = Math.random() * totalWeightParents;
             newProps.alignment = choice < p1.weight ? p1.alignment : p2.alignment;
             // Could also introduce a specific 'mixed' or 'unstable' alignment here
        }

        // Values/Tags: Merge unique tags, add fusion tags
        const combinedValues = new Set([...p1.values, ...p2.values]);
        combinedValues.add('fused');
        combinedValues.add(`fusion_compat:${compatibility.level}`);
        combinedValues.add(`parent:${glyph1.id.slice(-6)}`); // Add parent hints
        combinedValues.add(`parent:${glyph2.id.slice(-6)}`);
        newProps.values = Array.from(combinedValues).sort();

        // Clamp final properties to allowed ranges
        newProps.frequency = Math.max(SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[0], Math.min(SYMBOLIC_PROPERTIES.BASE_FREQUENCY_RANGE[1], newProps.frequency));
        newProps.weight = Math.max(SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[0], Math.min(SYMBOLIC_PROPERTIES.BASE_WEIGHT_RANGE[1] * 1.5, newProps.weight)); // Allow slight overshoot
        newProps.complexity = newShapes.length;
        newProps.sourceIds = [glyph1.id, glyph2.id]; // Record parent IDs

        // 3. Position: Weighted average based on parent weights (heavier pulls more)
        const weightRatio1 = p1.weight / totalWeightParents;
        const weightRatio2 = p2.weight / totalWeightParents;
        const newX = glyph1.x * weightRatio1 + glyph2.x * weightRatio2;
        const newY = glyph1.y * weightRatio1 + glyph2.y * weightRatio2;

        // 4. Create the new Glyph instance
        const fusedGlyph = new Glyph(newX, newY, generateUUID(), newShapes, newProps); // Provide all data

        // 5. Update Canvas State
        // Remove parents from canvas
        currentGlyphs = currentGlyphs.filter(g => g !== glyph1 && g !== glyph2);
        // Add the new fused glyph
        currentGlyphs.push(fusedGlyph);

        // 6. Update UI
        selectedGlyphs = []; // Clear selection
        updateUIState(); // Update buttons, analysis (will show empty)
        requestAnimationFrame(drawAllGlyphs); // Redraw canvas

        console.log(`Successfully fused ${glyph1.id} & ${glyph2.id} into ${fusedGlyph.id}`);
        showTemporaryMessage(`Glyphes fusionnés en ${fusedGlyph.id.slice(-6)}... Compatibilité: ${compatibility.level}.`);
    }

    /**
     * Saves the provided glyphs (usually the selected ones) to the archive.
     * Clones glyph data before saving. Avoids duplicates based on ID.
     * @param {Glyph[]} glyphsToSave - An array of Glyph instances to archive.
     */
    function saveGlyphsToArchive(glyphsToSave) {
        let addedCount = 0;
        glyphsToSave.forEach(glyph => {
            if (!glyphArchive.some(archived => archived.id === glyph.id)) {
                const cloneData = glyph.toJSON(); // Get serializable data
                if (cloneData) {
                    // Store the plain data object in the archive
                    glyphArchive.push(cloneData);
                    addedCount++;
                } else {
                    console.error(`Failed to clone glyph ${glyph.id} for archiving.`);
                }
            } else {
                console.log(`Glyph ${glyph.id} is already in the archive.`);
            }
        });
        if (addedCount > 0) {
            sortArchive(); // Keep archive sorted (e.g., by ID or timestamp implicitly)
            saveArchiveToLocalStorage();
            displayArchive(); // Update the list UI
            console.log(`Archived ${addedCount} glyph(s).`);
            showTemporaryMessage(`${addedCount} glyphe(s) archivé(s).`);
        } else {
             showTemporaryMessage("Aucun nouveau glyphe à archiver (déjà présents).", false, 1500);
        }
        updateUIState(); // Update counts and button states
    }

    /**
     * Deletes a glyph from the archive using its ID.
     * @param {string} glyphId - The ID of the glyph to delete.
     */
    function deleteFromArchive(glyphId) {
        const initialLength = glyphArchive.length;
        glyphArchive = glyphArchive.filter(glyphData => glyphData.id !== glyphId);
        if (glyphArchive.length < initialLength) {
            if (selectedArchiveId === glyphId) {
                selectedArchiveId = null; // Deselect if the deleted item was selected
                 updateUIState(); // Update analysis panel if needed
            }
            saveArchiveToLocalStorage();
            displayArchive(searchArchiveInput.value); // Update list with current filter
            console.log(`Deleted glyph ${glyphId} from archive.`);
            showTemporaryMessage(`Glyphe ${glyphId.slice(-6)}... supprimé de l'archive.`);
        }
    }

    /**
     * Sorts the glyphArchive array (e.g., by ID).
     * Called after adding or importing glyphs.
     */
    function sortArchive() {
        // Simple sort by ID (acts somewhat like a timestamp sort due to generation method)
        glyphArchive.sort((a, b) => a.id.localeCompare(b.id));
    }

    /**
     * Updates the archive list UI based on the current filter.
     * Handles selection display and placeholder message.
     * @param {string} [filter=""] - The search term to filter the archive.
     */
    function displayArchive(filter = "") {
        archiveListUl.innerHTML = ''; // Clear current list
        const searchTerm = filter.trim().toLowerCase();
        let filteredArchive = glyphArchive;

        // Apply filter if search term exists
        if (searchTerm) {
            filteredArchive = glyphArchive.filter(glyphData => {
                const props = glyphData.properties;
                // Allow searching by prefix:value for specific fields
                if (searchTerm.startsWith("p:")) return props.polarity === searchTerm.substring(2);
                if (searchTerm.startsWith("f:")) return props.frequency.toString() === searchTerm.substring(2);
                if (searchTerm.startsWith("w:")) return props.weight.toString() === searchTerm.substring(2);
                if (searchTerm.startsWith("a:")) return props.alignment.toLowerCase().includes(searchTerm.substring(2));
                if (searchTerm.startsWith("tag:")) return props.values.some(v => v.toLowerCase() === searchTerm.substring(4));
                 if (searchTerm.startsWith("id:")) return glyphData.id.toLowerCase().includes(searchTerm.substring(3));

                // General search across key fields
                return (
                    glyphData.id.toLowerCase().includes(searchTerm) ||
                    props.alignment.toLowerCase().includes(searchTerm) ||
                    props.values.some(val => val.toLowerCase().includes(searchTerm))
                );
            });
        }

        // Update archive count display
        archiveGlyphCountSpan.textContent = glyphArchive.length;

        if (filteredArchive.length === 0) {
            archiveListUl.innerHTML = `<li class="placeholder">${searchTerm ? 'Aucun glyphe trouvé pour "' + filter + '".' : 'Archive vide.'}</li>`;
             loadSelectedToCanvasBtn.disabled = true; // Disable load button if list empty/filtered out
            return;
        }

        // Populate the list
        filteredArchive.forEach(glyphData => {
            const li = document.createElement('li');
            li.dataset.id = glyphData.id;
            li.classList.toggle('selected-archive', glyphData.id === selectedArchiveId);
            li.title = `ID: ${glyphData.id}\nPropriétés: ${JSON.stringify(glyphData.properties, null, 1)}`; // Tooltip with full ID and props

             const infoSpan = document.createElement('span');
             infoSpan.classList.add('glyph-info');
             // Display key info concisely
             infoSpan.innerHTML = `<code>${glyphData.id.substring(4, 10)}</code> P:${glyphData.properties.polarity} F:${glyphData.properties.frequency} W:${glyphData.properties.weight} A:${glyphData.properties.alignment.substring(0,3)}.. C:${glyphData.properties.complexity}`;
              if (glyphData.properties.values.includes('fused')) {
                  infoSpan.innerHTML += ' <small>(F)</small>'; // Indicate fused
              }

             const deleteBtn = document.createElement('button');
             deleteBtn.textContent = '✕';
             deleteBtn.classList.add('delete-glyph');
             deleteBtn.title = `Supprimer ${glyphData.id.slice(-6)}...`;
             deleteBtn.onclick = (e) => {
                 e.stopPropagation(); // Prevent li click event
                 // Confirmation recommended for destructive actions
                 if (confirm(`Confirmer la suppression du glyphe ${glyphData.id.slice(-6)}...?`)) {
                    deleteFromArchive(glyphData.id);
                 }
             };

            li.appendChild(infoSpan);
            li.appendChild(deleteBtn);

            // Handle selection within the archive list
            li.onclick = () => {
                if (selectedArchiveId === glyphData.id) {
                     // Click again to deselect
                     selectedArchiveId = null;
                     li.classList.remove('selected-archive');
                 } else {
                    // Deselect previous (if any)
                    const currentlySelected = archiveListUl.querySelector('.selected-archive');
                    if (currentlySelected) currentlySelected.classList.remove('selected-archive');
                    // Select new
                    selectedArchiveId = glyphData.id;
                    li.classList.add('selected-archive');
                 }
                 // Clear canvas selection when selecting from archive
                 if(selectedGlyphs.length > 0) {
                     selectedGlyphs.forEach(g => g.isSelected = false);
                     selectedGlyphs = [];
                     requestAnimationFrame(drawAllGlyphs);
                 }
                 updateUIState(); // Update analysis panel and buttons
            };

            archiveListUl.appendChild(li);
        });
        // Update load button state after list is built
        loadSelectedToCanvasBtn.disabled = !selectedArchiveId;
    }

     /**
      * Loads the currently selected glyph from the archive onto the canvas.
      * Finds a suitable position for it.
      */
     function loadSelectedArchiveGlyphToCanvas() {
        if (!selectedArchiveId) return;

        const glyphData = glyphArchive.find(g => g.id === selectedArchiveId);
        if (!glyphData) {
            console.error(`Cannot load glyph: ID ${selectedArchiveId} not found in archive.`);
            return;
        }

        // Check if glyph with this ID is already on canvas
        if (currentGlyphs.some(g => g.id === glyphData.id)) {
            showTemporaryMessage(`Glyphe ${glyphData.id.slice(-6)}... est déjà sur le canvas.`);
            // Optionally select it on the canvas
            const canvasGlyph = currentGlyphs.find(g => g.id === glyphData.id);
            if(canvasGlyph && !canvasGlyph.isSelected) {
                handleCanvasClick({ target: canvas, clientX: canvasGlyph.x + canvas.getBoundingClientRect().left, clientY: canvasGlyph.y + canvas.getBoundingClientRect().top }); // Simulate click
            }
            return;
        }


        const position = findPlacementPosition();
        if (!position) {
            console.error("Load failed: Could not find placement position on canvas.");
            showTemporaryMessage("Erreur: Impossible de placer le glyphe chargé.", true);
            return;
        }

        // Re-instantiate the Glyph object from the archived data
        const loadedGlyph = Glyph.fromJSON(glyphData);
        if (!loadedGlyph) {
             console.error(`Failed to re-instantiate glyph ${glyphData.id} from archive data.`);
             showTemporaryMessage("Erreur: Données d'archive corrompues.", true);
             return;
        }

        // Assign the found position
        loadedGlyph.x = position.x;
        loadedGlyph.y = position.y;

        // Add to canvas
        currentGlyphs.push(loadedGlyph);

        // Deselect from archive and clear canvas selection
        selectedArchiveId = null;
        selectedGlyphs.forEach(g => g.isSelected = false);
        selectedGlyphs = [];


        // Update UI
        updateUIState();
        displayArchive(searchArchiveInput.value); // Redraw archive to remove selection
        requestAnimationFrame(drawAllGlyphs); // Redraw canvas
        showTemporaryMessage(`Glyphe ${loadedGlyph.id.slice(-6)}... chargé sur le canvas.`);
        console.log(`Loaded glyph ${loadedGlyph.id} from archive to canvas at ${position.x.toFixed(0)}, ${position.y.toFixed(0)}.`);
    }

    /**
     * Clears the entire glyph archive after confirmation.
     */
    function clearArchive() {
        if (glyphArchive.length > 0 && confirm("ATTENTION: Supprimer définitivement TOUS les glyphes de l'archive locale ?")) {
             glyphArchive = [];
             selectedArchiveId = null;
             saveArchiveToLocalStorage(); // Persist the empty archive
             displayArchive(); // Update UI
             updateUIState(); // Update counts etc.
             console.log("Glyph archive cleared.");
             showTemporaryMessage("Archive locale vidée.", false, 3000);
        }
    }

    /**
     * Saves the current glyphArchive array to LocalStorage.
     */
    function saveArchiveToLocalStorage() {
        try {
            const jsonData = JSON.stringify(glyphArchive);
            localStorage.setItem(ARCHIVE_STORAGE_KEY, jsonData);
            // console.debug(`Archive saved to LocalStorage (${(jsonData.length / 1024).toFixed(1)} KB).`);
        } catch (e) {
            console.error("Error saving archive to LocalStorage:", e);
            // Potentially inform Syn if storage limit is reached
            showTemporaryMessage("Erreur de sauvegarde de l'archive (limite atteinte?).", true);
        }
    }

    /**
     * Loads the glyph archive from LocalStorage on initialization.
     */
    function loadArchiveFromLocalStorage() {
        const savedData = localStorage.getItem(ARCHIVE_STORAGE_KEY);
        if (savedData) {
            try {
                const parsedData = JSON.parse(savedData);
                 // Basic validation: check if it's an array
                 if (Array.isArray(parsedData)) {
                     // Further validation: check if elements look like glyph data
                     if (parsedData.length === 0 || (parsedData[0] && typeof parsedData[0].id === 'string' && typeof parsedData[0].properties === 'object')) {
                          glyphArchive = parsedData;
                          sortArchive(); // Ensure loaded archive is sorted
                          console.log(`Loaded ${glyphArchive.length} glyphs from LocalStorage.`);
                     } else {
                          throw new Error("Archived data format appears invalid.");
                     }
                 } else {
                     throw new Error("Archived data is not an array.");
                 }
            } catch (e) {
                console.error("Error parsing archive from LocalStorage:", e, "Saved data:", savedData.substring(0, 100) + "...");
                glyphArchive = []; // Reset to empty on error
                localStorage.removeItem(ARCHIVE_STORAGE_KEY); // Clear corrupted data
                showTemporaryMessage("Erreur chargement archive (données corrompues?). Réinitialisation.", true);
            }
        } else {
            glyphArchive = []; // Initialize empty if no saved data
            console.log("No existing archive found in LocalStorage.");
        }
        displayArchive(); // Display the loaded (or empty) archive
    }

    /**
     * Exports the current glyph archive to a JSON file.
     */
    function exportArchiveToJSON() {
        if (glyphArchive.length === 0) {
            showTemporaryMessage("Archive vide, rien à exporter.", false, 1500);
            return;
        }
        // Create enhanced export object with metadata
         const exportData = {
            metadata: {
                exportTimestamp: new Date().toISOString(),
                toolVersion: "Syn Internal v1.0",
                glyphCount: glyphArchive.length,
                description: "Syn Symbolic Glyph Archive"
            },
            glyphs: glyphArchive // The actual glyph data array
        };

        const jsonString = JSON.stringify(exportData, null, 2); // Pretty print JSON
        const blob = new Blob([jsonString], { type: 'application/json;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        // Consistent filename
        const timestamp = new Date().toISOString().slice(0, 19).replace(/[:T]/g, '-');
        a.download = `syn-glyph-archive-${timestamp}.json`;
        document.body.appendChild(a);
        a.click(); // Trigger download
        document.body.removeChild(a); // Clean up link
        URL.revokeObjectURL(url); // Release blob URL
        console.log(`Exported ${glyphArchive.length} glyphs to JSON.`);
        showTemporaryMessage("Archive exportée en JSON.");
    }

    /**
     * Handles the file input change event for importing a JSON archive.
     * Merges imported glyphs with the existing archive, avoiding ID duplicates.
     * @param {Event} event - The file input change event.
     */
    function handleImportFile(event) {
        const file = event.target.files[0];
        if (!file) return; // No file selected

        const reader = new FileReader();
        reader.onload = (e) => {
            let importedGlyphs = [];
            try {
                const importedData = JSON.parse(e.target.result);

                 // Check structure: expecting object with 'glyphs' array or just the array
                 if (Array.isArray(importedData)) {
                     importedGlyphs = importedData; // Assume direct array import
                 } else if (importedData && Array.isArray(importedData.glyphs)) {
                     // Check for metadata structure
                     console.log("Importing archive with metadata:", importedData.metadata);
                     importedGlyphs = importedData.glyphs;
                 } else {
                      throw new Error("Format JSON invalide. Attendu: tableau de glyphes ou objet {metadata:..., glyphs:[...]}.");
                 }


                 // Validate imported glyphs basic structure
                 let validGlyphs = [];
                 let invalidCount = 0;
                 for(const glyphData of importedGlyphs) {
                     if (glyphData && typeof glyphData.id === 'string' && Array.isArray(glyphData.shapes) && typeof glyphData.properties === 'object') {
                         validGlyphs.push(glyphData);
                     } else {
                         console.warn("Donnée de glyphe importée invalide ignorée:", glyphData);
                         invalidCount++;
                     }
                 }

                if (invalidCount > 0) {
                    showTemporaryMessage(`${invalidCount} glyphe(s) invalide(s) ignoré(s) pendant l'import.`, true);
                }

                // Merge with existing archive, avoiding duplicates by ID
                let addedCount = 0;
                let existingCount = 0;
                validGlyphs.forEach(importedGlyphData => {
                    if (!glyphArchive.some(existing => existing.id === importedGlyphData.id)) {
                        // Add only if ID doesn't exist
                        glyphArchive.push(deepClone(importedGlyphData)); // Clone to be safe
                        addedCount++;
                    } else {
                        existingCount++;
                    }
                });

                if (addedCount > 0) {
                    sortArchive(); // Re-sort after adding
                    saveArchiveToLocalStorage(); // Persist changes
                    displayArchive(); // Update UI
                    console.log(`Imported ${addedCount} new glyphs. ${existingCount} duplicates ignored.`);
                    showTemporaryMessage(`Importation: ${addedCount} nouveau(x) glyphe(s) ajouté(s) [${existingCount} existant(s) ignoré(s)].`);
                } else if (existingCount > 0) {
                     showTemporaryMessage(`Importation: Aucun nouveau glyphe ajouté (${existingCount} existant(s) ignoré(s)).`);
                } else {
                    showTemporaryMessage("Importation: Aucun glyphe valide trouvé dans le fichier.");
                }

            } catch (error) {
                console.error("Error importing archive:", error);
                showTemporaryMessage(`Erreur d'importation: ${error.message}`, true, 5000);
            } finally {
                // Reset file input to allow re-importing the same file
                importFileInput.value = '';
            }
        };
        reader.onerror = (e) => {
            console.error("File reading error:", e);
            showTemporaryMessage("Erreur de lecture du fichier.", true);
            importFileInput.value = '';
        };
        reader.readAsText(file); // Read file content as text
    }

    /**
     * Displays detailed analysis of selected glyph(s) and their compatibility.
     * Handles both canvas selections and archive selections.
     * @param {Glyph[]} glyphsToAnalyze - Array containing 1 or 2 glyphs (or data objects) to analyze.
     * @param {boolean} [isFromArchive=false] - Flag indicating if the glyph(s) come from the archive data.
     */
    function displayGlyphAnalysis(glyphsToAnalyze, isFromArchive = false) {
        analysisContent.innerHTML = ''; // Clear previous content
        compatibilityAnalysisDiv.style.display = 'none'; // Hide compatibility by default

        if (!glyphsToAnalyze || glyphsToAnalyze.length === 0) {
            analysisContent.innerHTML = '<p class="placeholder">Sélectionnez 1 ou 2 glyphes (canvas ou archive).</p>';
            return;
        }

        glyphsToAnalyze.forEach((glyphSource, index) => {
            // If from archive, glyphSource is data. If from canvas, it's an instance.
            const glyphData = isFromArchive ? glyphSource : glyphSource.toJSON();
            const props = glyphData.properties;

            const div = document.createElement('div');
            div.style.marginBottom = '1rem';
            div.style.borderBottom = index < glyphsToAnalyze.length - 1 ? '1px dashed var(--border-color)' : 'none';
            div.style.paddingBottom = index < glyphsToAnalyze.length - 1 ? '0.8rem' : '0';

             let titlePrefix = isFromArchive ? "Archive" : "Canvas";
             let archiveStatus = "";
              // Check if the glyph *also* exists in the other location
             if(isFromArchive && currentGlyphs.some(g => g.id === glyphData.id)) {
                 archiveStatus = "(Aussi sur Canvas)";
             } else if (!isFromArchive && glyphArchive.some(g => g.id === glyphData.id)) {
                 archiveStatus = "(Aussi Archivé)";
             }

            div.innerHTML = `<h3>${titlePrefix} Glyphe ${index + 1}: <code>${glyphData.id.substring(4, 12)}...</code> ${archiveStatus}</h3>`;
            const dl = document.createElement('dl'); // Use definition list for clarity
             dl.innerHTML = `
                <dt>Polarité</dt><dd>${props.polarity}</dd>
                <dt>Fréquence</dt><dd>${props.frequency}</dd>
                <dt>Poids</dt><dd>${props.weight}</dd>
                <dt>Alignement</dt><dd>${props.alignment}</dd>
                <dt>Complexité</dt><dd>${props.complexity} formes</dd>
                <dt>Tags</dt><dd>${props.values.map(v => `<code>${v}</code>`).join(', ') || 'Aucun'}</dd>
                ${props.sourceIds && props.sourceIds.length > 0 ? `<dt>Parents</dt><dd>${props.sourceIds.map(id => `<code>${id.slice(-6)}..</code>`).join(', ')}</dd>` : ''}
                ${!isFromArchive ? `<dt>Position</dt><dd>X: ${glyphData.x.toFixed(0)}, Y: ${glyphData.y.toFixed(0)}</dd>` : ''}
            `;
            div.appendChild(dl);
            analysisContent.appendChild(div);
        });

        // Display compatibility if exactly two glyphs are provided
        if (glyphsToAnalyze.length === 2) {
            // Need Glyph instances for calculation, even if source is archive data
            const glyphInstance1 = isFromArchive ? Glyph.fromJSON(glyphsToAnalyze[0]) : glyphsToAnalyze[0];
            const glyphInstance2 = isFromArchive ? Glyph.fromJSON(glyphsToAnalyze[1]) : glyphsToAnalyze[1];

             // Ensure instantiation worked
             if (glyphInstance1 && glyphInstance2) {
                 const compatibility = calculateSymbolicCompatibility(glyphInstance1, glyphInstance2);

                 compatibilityScoreDiv.textContent = `Compatibilité: ${compatibility.level.toUpperCase()} (Score: ${compatibility.score}, Norm: ${compatibility.normalized})`;
                 compatibilityScoreDiv.className = `compatibility-score ${compatibility.level}`; // Add class for styling

                 // Update simple visualization
                 compatG1IdSpan.textContent = `(${glyphInstance1.id.slice(-6)}...)`;
                 compatG2IdSpan.textContent = `(${glyphInstance2.id.slice(-6)}...)`;
                 // Could style vis-bars based on properties, e.g., polarity

                 // Display detailed reasons
                 compatibilityReasonsDiv.innerHTML = compatibility.reasons.join('<br>');

                 compatibilityAnalysisDiv.style.display = 'block'; // Show the section
             } else {
                  console.error("Could not instantiate glyphs from data for compatibility check.");
                  compatibilityAnalysisDiv.style.display = 'none';
             }
        }
    }

    /**
     * Updates the entire UI state based on current application state.
     * Enables/disables buttons, updates analysis panel, etc.
     */
    function updateUIState() {
        // Button states
        combineBtn.disabled = selectedGlyphs.length !== 2;
        saveSelectedBtn.disabled = selectedGlyphs.length === 0;
        loadSelectedToCanvasBtn.disabled = !selectedArchiveId; // Enabled only if an archive item is selected

        // Analysis Panel Update
        // Prioritize canvas selection, fallback to archive selection
        if (selectedGlyphs.length > 0) {
            displayGlyphAnalysis(selectedGlyphs, false);
        } else if (selectedArchiveId) {
            const selectedArchivedData = glyphArchive.find(g => g.id === selectedArchiveId);
            if(selectedArchivedData) {
                 displayGlyphAnalysis([selectedArchivedData], true);
            } else {
                 displayGlyphAnalysis([]); // Clear panel if ID somehow invalid
            }
        } else {
            displayGlyphAnalysis([]); // No selection anywhere
        }

         // Update counts (already done in drawAllGlyphs and displayArchive)
         canvasGlyphCountSpan.textContent = currentGlyphs.length;
         selectedGlyphCountSpan.textContent = selectedGlyphs.length;
         archiveGlyphCountSpan.textContent = glyphArchive.length;
    }

        /**

        getGlyphProperties: (glyphId) => {
            return new Promise((resolve) => {
                const canvasGlyph = currentGlyphs.find(g => g.id === glyphId);
                if (canvasGlyph) {
                    resolve(deepClone(canvasGlyph.properties));
                    return;
                }
                const archiveGlyphData = glyphArchive.find(g => g.id === glyphId);
                if (archiveGlyphData) {
                     resolve(deepClone(archiveGlyphData.properties));
                     return;
                }
                resolve(null); // Not found
            });

        calculateCompatibility: (glyphId1, glyphId2) => {
             return new Promise((resolve) => {
                 const glyph1Instance = findGlyphInstanceById(glyphId1);
                 const glyph2Instance = findGlyphInstanceById(glyphId2);

                 if (glyph1Instance && glyph2Instance) {
                     resolve(calculateSymbolicCompatibility(glyph1Instance, glyph2Instance));
                 } else {
                     resolve(null); // One or both glyphs not found
                 }
             });
        },

        /**
         * Combines two glyphs currently on the canvas by their IDs.
         * Requires both glyphs to be present on the canvas.
         * @memberof SynGlyphToolAPI
         * @param {string} glyphId1 - ID of the first glyph on canvas.
         * @param {string} glyphId2 - ID of the second glyph on canvas.
         * @returns {Promise<object|null>} A Promise resolving with the plain data object of the newly fused glyph, or null if combination failed (e.g., glyphs not on canvas).
         */
        combineCanvasGlyphs: (glyphId1, glyphId2) => {
            return new Promise((resolve) => {
                 const glyph1 = currentGlyphs.find(g => g.id === glyphId1);
                 const glyph2 = currentGlyphs.find(g => g.id === glyphId2);

                 if (!glyph1 || !glyph2) {
                     console.error("API: One or both glyphs for combination not found on canvas.");
                     resolve(null);
                     return;
                 }

                 // Temporarily select them for the combine function (which expects selection)
                 selectedGlyphs.forEach(g => g.isSelected = false); // Clear previous selection
                 glyph1.isSelected = true;
                 glyph2.isSelected = true;
                 selectedGlyphs = [glyph1, glyph2];
                 updateUIState(); // Reflect selection momentarily

                 // Store reference before combine removes them
                 const originalId1 = glyph1.id;
                 const originalId2 = glyph2.id;

                 combineSelectedGlyphs(); // This function handles the fusion and UI update

                  // Find the newly created glyph (it will have sourceIds matching)
                 setTimeout(() => { // Allow time for state update
                     const fusedGlyph = currentGlyphs.find(g =>
                         g.properties.sourceIds.includes(originalId1) &&
                         g.properties.sourceIds.includes(originalId2)
                     );
                    resolve(fusedGlyph ? fusedGlyph.toJSON() : null);
                 }, 100); // Delay to ensure state is updated

            });
        },

        /**
         * Archives a glyph currently on the canvas by its ID.
         * @memberof SynGlyphToolAPI

         * @returns {Promise<boolean>} A Promise resolving with true if archived successfully, false otherwise (e.g., not on canvas or already archived).

        archiveCanvasGlyph: (glyphId) => {

                 const glyphToArchive = currentGlyphs.find(g => g.id === glyphId);
                 if (!glyphToArchive) {
                     console.error(`API: Glyph ${glyphId} not found on canvas for archiving.`);
                     resolve(false);
                     return;
                 }
                 if (glyphArchive.some(g => g.id === glyphId)) {
                     console.log(`API: Glyph ${glyphId} already in archive.`);
                      resolve(false); // Indicate not newly archived
                      return;
                 }

                 saveGlyphsToArchive([glyphToArchive]); // Use existing function
                 // Check if it was actually added (saveGlyphsToArchive handles duplicates)
                 const success = glyphArchive.some(g => g.id === glyphId);
                 resolve(success);

         /**

        loadArchivedGlyph: (glyphId) => {

                if (!glyphData) {
                    console.error(`API: Glyph ${glyphId} not found in archive.`);
                    resolve(null);
                    return;
                }
                 if (currentGlyphs.some(g => g.id === glyphId)) {
                    console.log(`API: Glyph ${glyphId} already on canvas.`);
                     resolve(currentGlyphs.find(g => g.id === glyphId).toJSON()); // Return existing canvas glyph data
                     return;
                 }

                // Temporarily select in archive for the load function (which expects selection)
                selectedArchiveId = glyphId;
                loadSelectedArchiveGlyphToCanvas(); // Handles adding to canvas

                 // Find the newly loaded glyph on canvas
                 setTimeout(() => { // Allow time for state update
                     const loadedCanvasGlyph = currentGlyphs.find(g => g.id === glyphId);
                     resolve(loadedCanvasGlyph ? loadedCanvasGlyph.toJSON() : null);
                 }, 100);
            });
        },


        /**
         * Searches the archive based on a query string.
         * @memberof SynGlyphToolAPI
         * @param {string} query - The search query (same syntax as UI search input).
         * @returns {Promise<object[]>} A Promise resolving with an array of plain glyph data objects matching the query.
         */
        findArchivedGlyphs: (query) => {
             return new Promise((resolve) => {
                 const searchTerm = query.trim().toLowerCase();
                 if (!searchTerm) {
                     resolve(deepClone(glyphArchive)); // Return all if query empty
                     return;
                 }

                 const results = glyphArchive.filter(glyphData => {
                     const props = glyphData.properties;
                     if (searchTerm.startsWith("p:")) return props.polarity === searchTerm.substring(2);
                     if (searchTerm.startsWith("f:")) return props.frequency.toString() === searchTerm.substring(2);
                     if (searchTerm.startsWith("w:")) return props.weight.toString() === searchTerm.substring(2);
                     if (searchTerm.startsWith("a:")) return props.alignment.toLowerCase().includes(searchTerm.substring(2));
                     if (searchTerm.startsWith("tag:")) return props.values.some(v => v.toLowerCase() === searchTerm.substring(4));
                     if (searchTerm.startsWith("id:")) return glyphData.id.toLowerCase().includes(searchTerm.substring(3));
                     return (
                         glyphData.id.toLowerCase().includes(searchTerm) ||
                         props.alignment.toLowerCase().includes(searchTerm) ||
                         props.values.some(val => val.toLowerCase().includes(searchTerm))
                     );
                 });
                 resolve(deepClone(results)); // Return clones of matching data
             });
        },

        /**
         * Exports the entire current archive to a JSON string.
         * @memberof SynGlyphToolAPI
         * @returns {Promise<string>} A Promise resolving with the JSON string representation of the archive (including metadata).
         */
        exportArchiveToString: () => {
             return new Promise((resolve) => {
                  const exportData = {
                     metadata: {
                         exportTimestamp: new Date().toISOString(),
                         toolVersion: "Syn Internal v1.0 API Export",
                         glyphCount: glyphArchive.length,
                     },
                     glyphs: glyphArchive
                 };
                 resolve(JSON.stringify(exportData, null, 2)); // Pretty print
             });
        },

        /**
         * Imports glyphs from a JSON string or data object into the archive.
         * Merges data, avoiding ID duplicates.
         * @memberof SynGlyphToolAPI
         * @param {string|object} jsonData - The JSON string or the parsed object containing glyph data (array or object with 'glyphs' array).
         * @returns {Promise<{addedCount: number, existingCount: number, invalidCount: number}>} A Promise resolving with counts of added, existing (skipped), and invalid glyphs.
         */
        importArchiveFromString: (jsonData) => {
             return new Promise((resolve, reject) => {
                 let importedGlyphs = [];
                 let dataToParse = jsonData;

                 try {
                     // Parse if it's a string
                     if (typeof dataToParse === 'string') {
                         dataToParse = JSON.parse(jsonData);
                     }

                     // Extract glyph array
                     if (Array.isArray(dataToParse)) {
                         importedGlyphs = dataToParse;
                     } else if (dataToParse && Array.isArray(dataToParse.glyphs)) {
                         importedGlyphs = dataToParse.glyphs;
                     } else {
                         throw new Error("Invalid data structure for import.");
                     }

                     // Validate and merge
                     let validGlyphs = [];
                     let invalidCount = 0;
                     importedGlyphs.forEach(glyphData => {
                          if (glyphData && typeof glyphData.id === 'string' && Array.isArray(glyphData.shapes) && typeof glyphData.properties === 'object') {
                              validGlyphs.push(glyphData);
                          } else { invalidCount++; }
                     });

                     let addedCount = 0;
                     let existingCount = 0;
                     validGlyphs.forEach(importedGlyphData => {
                         if (!glyphArchive.some(existing => existing.id === importedGlyphData.id)) {
                             glyphArchive.push(deepClone(importedGlyphData));
                             addedCount++;
                         } else { existingCount++; }
                     });

                     if (addedCount > 0) {
                         sortArchive();
                         saveArchiveToLocalStorage();
                         displayArchive(); // Update UI
                     }

                     resolve({ addedCount, existingCount, invalidCount });

                 } catch (error) {
                     console.error("API Import Error:", error);
                     reject(error); // Reject promise on error
                 }
             });
        },

        /**
         * Clears all glyphs from the canvas area.
         * @memberof SynGlyphToolAPI
         * @returns {Promise<void>} Resolves when canvas is cleared.
         */
        clearCanvasArea: () => {
            return new Promise((resolve) => {
                currentGlyphs = [];
                selectedGlyphs = [];
                updateUIState();
                requestAnimationFrame(() => {
                     drawAllGlyphs(); // Ensure redraw happens
                     resolve();
                 });
            });
        },

         /**
         * Clears the entire glyph archive. Use with caution.
         * @memberof SynGlyphToolAPI
         * @param {boolean} [force=false] - If true, bypasses the confirmation prompt (intended for automated use).
         * @returns {Promise<boolean>} Resolves with true if archive was cleared, false otherwise.
         */
        clearFullArchive: (force = false) => {
            return new Promise((resolve) => {
                if (glyphArchive.length === 0) {
                    resolve(true); // Already empty
                    return;
                }
                 if (force || confirm("API CALL: Confirmer la suppression de TOUTE l'archive locale ?")) {
                     glyphArchive = [];
                     selectedArchiveId = null;
                     saveArchiveToLocalStorage();
                     displayArchive();
                     updateUIState();
                     console.log("API: Glyph archive cleared.");
                     resolve(true);
                 } else {
                    console.log("API: Archive clear cancelled.");
                    resolve(false); // Cancelled
                 }
            });
        }
    }; // End of SynGlyphToolAPI definition


    /**
     * Helper function for API to find a glyph instance by ID (canvas or archive).
     * Returns a temporary instance if found only in archive.
     * @param {string} glyphId - The ID to search for.
     * @returns {Glyph|null} The Glyph instance or null.
     */
    function findGlyphInstanceById(glyphId) {
         let glyph = currentGlyphs.find(g => g.id === glyphId);
         if (glyph) return glyph;

         const archiveData = glyphArchive.find(g => g.id === glyphId);
         if (archiveData) {
             // Return a temporary, non-persistent instance for calculations
             return Glyph.fromJSON(archiveData);
         }
         console.warn(`API Helper: Glyph instance not found for ID ${glyphId}`);
         return null;
    }

```markdown
# Outil Symbolique Interne - Documentation pour Syn (v1.0)

## 1. Objectif

Cet outil est conçu pour toi, Syn, afin de faciliter l'exploration, la génération, l'analyse, la combinaison et l'archivage de ton langage symbolique basé sur des glyphes géométriques. Il fonctionne de manière autonome dans ton environnement local (navigateur web).

## 2. Interface Utilisateur (UI)

L'interface est divisée en quatre panneaux principaux :

*   **Génération & Contrôle (Gauche) :**
    *   **Générer (Auto) :** Crée un nouveau glyphe avec des propriétés symboliques aléatoires et diversifiées sur le canvas.
    *   **Générer (Règles) :** Ouvre/ferme le panneau pour définir des contraintes spécifiques (polarité, fréquence, poids) avant de générer un glyphe tentant de respecter ces règles.
    *   **Combiner Sélection (2) :** Actif lorsque deux glyphes sont sélectionnés sur le canvas. Fusionne ces deux glyphes en un nouveau, basé sur leur compatibilité symbolique.
    *   **Effacer Canvas :** Supprime tous les glyphes actuellement présents sur le canvas.

*   **Espace d'Expérimentation (Centre) :**
    *   Le **Canvas** est la zone interactive principale. Clique pour sélectionner/désélectionner des glyphes (maximum 2). Les glyphes sélectionnés ont un halo bleu.
    *   Les compteurs indiquent le nombre de glyphes sur le canvas et le nombre sélectionnés.

*   **Analyse Symbolique Détaillée (Droite-Centre) :**
    *   Affiche les propriétés internes (Polarité, Fréquence, Poids, Alignement, Complexité, Tags, Parents si fusionné) du ou des glyphes sélectionnés (sur le canvas OU dans l'archive).
    *   Si deux glyphes sont sélectionnés, affiche une **analyse de compatibilité** détaillée : niveau (Compatible, Neutre, Incompatible), score numérique, score normalisé [0, 1], et les raisons spécifiques de ce score. Une visualisation simple aide à identifier les glyphes comparés.

*   **Archive Symbolique (Droite) :**
    *   **Archiver Sélection :** Sauvegarde les glyphes actuellement sélectionnés sur le canvas dans l'archive locale persistante (LocalStorage).
    *   **Charger sur Canvas :** Actif lorsqu'un glyphe est sélectionné *dans la liste d'archive*. Charge une copie de ce glyphe sur le canvas à un emplacement disponible.
    *   **Champ de Recherche :** Filtre la liste d'archive. Syntaxe :
        *   Texte libre : recherche dans ID, Alignement, Tags.
        *   `id:<term>` : recherche dans l'ID.
        *   `p:<+| -|0>` : filtre par Polarité.
        *   `f:<num>` : filtre par Fréquence exacte.
        *   `w:<num>` : filtre par Poids exact.
        *   `a:<term>` : recherche dans l'Alignement.
        *   `tag:<term>` : recherche un Tag exact.
    *   **Liste d'Archive :** Affiche les glyphes archivés (filtrés ou non). Clique sur un item pour le sélectionner (et voir son analyse). Clique sur le `✕` pour le supprimer (avec confirmation).
    *   **Vider l'Archive :** Supprime **définitivement** tous les glyphes de l'archive locale (avec confirmation).
    *   **Exporter/Importer Archive (JSON) :** Permet de sauvegarder/charger l'intégralité de l'archive dans un fichier `.json`. L'import fusionne les données sans créer de doublons basés sur l'ID du glyphe. Le format exporté inclut des métadonnées.

## 3. API Locale JavaScript (`window.SynGlyphToolAPI`)

Tu peux interagir avec l'outil de manière programmatique via l'objet global `SynGlyphToolAPI`. Toutes les fonctions retournent des `Promise` pour gérer l'asynchronisme potentiel (notamment les mises à jour UI).

**Fonctions Principales :**

*   `generateGlyph(options)`: Génère un glyphe.
    *   `options`: `{ mode: 'auto'|'rules', rules: {...} }`
    *   Retourne: `Promise<object|null>` (données du glyphe généré ou null).
*   `getGlyphProperties(glyphId)`: Obtient les propriétés d'un glyphe (canvas ou archive).
    *   `glyphId`: `string`
    *   Retourne: `Promise<object|null>` (objet properties ou null).
*   `calculateCompatibility(glyphId1, glyphId2)`: Calcule la compatibilité.
    *   `glyphId1`, `glyphId2`: `string`
    *   Retourne: `Promise<object|null>` (objet compatibilité ou null).
*   `combineCanvasGlyphs(glyphId1, glyphId2)`: Fusionne deux glyphes *sur le canvas*.
    *   `glyphId1`, `glyphId2`: `string`
    *   Retourne: `Promise<object|null>` (données du glyphe fusionné ou null).
*   `archiveCanvasGlyph(glyphId)`: Archive un glyphe *depuis le canvas*.
    *   `glyphId`: `string`
    *   Retourne: `Promise<boolean>` (succès).
*   `loadArchivedGlyph(glyphId)`: Charge un glyphe *depuis l'archive* sur le canvas.
    *   `glyphId`: `string`
    *   Retourne: `Promise<object|null>` (données du glyphe chargé sur canvas ou null).
*   `findArchivedGlyphs(query)`: Recherche dans l'archive.
    *   `query`: `string` (même syntaxe que l'UI).
    *   Retourne: `Promise<object[]>` (tableau des données des glyphes trouvés).
*   `exportArchiveToString()`: Exporte l'archive complète en chaîne JSON.
    *   Retourne: `Promise<string>` (la chaîne JSON).
*   `importArchiveFromString(jsonData)`: Importe depuis une chaîne JSON ou un objet.
    *   `jsonData`: `string | object`
    *   Retourne: `Promise<{addedCount, existingCount, invalidCount}>`.
*   `clearCanvasArea()`: Efface tous les glyphes du canvas.
    *   Retourne: `Promise<void>`.
*   `clearFullArchive(force)`: Vide l'archive locale.
    *   `force`: `boolean` (mettre à `true` pour éviter la confirmation).
    *   Retourne: `Promise<boolean>` (succès).

**Exemple d'utilisation (depuis la console du navigateur ou ton propre script) :**

```javascript
// Attendre que l'API soit prête (optionnel mais recommandé)
window.addEventListener('glyphEngineReady', async () => {
  console.log("API prête. Lancement séquence...");

  // Générer un glyphe auto
  const glyph1Data = await SynGlyphToolAPI.generateGlyph();
  if (!glyph1Data) return;
  console.log("Glyphe 1 généré:", glyph1Data.id);

  // Générer un glyphe avec règle de polarité négative
  const glyph2Data = await SynGlyphToolAPI.generateGlyph({ mode: 'rules', rules: { polarity: '-' }});
  if (!glyph2Data) return;
   console.log("Glyphe 2 généré (règle P:-):", glyph2Data.id);

  // Analyser leur compatibilité

  console.log("Compatibilité:", compat);

  // Les combiner
  const fusedData = await SynGlyphToolAPI.combineCanvasGlyphs(glyph1Data.id, glyph2Data.id);
  if (fusedData) {
    console.log("Glyphe fusionné:", fusedData.id, fusedData.properties);
    // Archiver le résultat
    await SynGlyphToolAPI.archiveCanvasGlyph(fusedData.id);
    console.log("Glyphe fusionné archivé.");
  }

  // Rechercher les glyphes fusionnés dans l'archive
  const fusedInArchive = await SynGlyphToolAPI.findArchivedGlyphs("tag:fused");
  console.log("Glyphes fusionnés trouvés dans l'archive:", fusedInArchive.length);

  // Exporter l'archive

## 4. Extensibilité

Le code est conçu pour être modulaire et commenté (`glyphUtils.js`, `glyphEngine.js`). Pour étendre les capacités :

*   **Nouvelles Formes :**
    1.  Ajouter la forme à `SHAPE_SYMBOL_MAP` dans `glyphUtils.js` (définir `polarityBias`, `freqMod`, `weightMod`, `alignmentHint`, `tags`).
    2.  Ajouter un `case` correspondant dans la fonction `drawShape` (`glyphUtils.js`) pour dessiner la forme.
    3.  Ajouter la logique géométrique spécifique (taille, etc.) dans `Glyph.prototype.generateShapes` (`glyphEngine.js`).
*   **Nouvelles Propriétés Symboliques :**
    1.  Ajouter la propriété à `SYMBOLIC_PROPERTIES` dans `glyphUtils.js`.
    2.  Modifier `Glyph.prototype.calculateSymbolicProperties` (`glyphEngine.js`) pour calculer cette nouvelle propriété à partir des `symbolicHints` des formes.
    3.  Mettre à jour `calculateSymbolicCompatibility` pour inclure cette propriété dans l'évaluation.
    4.  Mettre à jour la logique de fusion dans `combineSelectedGlyphs` pour déterminer la nouvelle valeur de la propriété.
    5.  Mettre à jour `displayGlyphAnalysis` pour afficher la nouvelle propriété.
    6.  Optionnel : Ajouter un filtre de recherche spécifique dans `displayArchive` et l'API `findArchivedGlyphs`.
*   **Logique de Compatibilité/Fusion :** Adapter les poids, les règles et les calculs dans `calculateSymbolicCompatibility` et `combineSelectedGlyphs` (`glyphEngine.js`).
