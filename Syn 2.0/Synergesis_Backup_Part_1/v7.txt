def _get_dominant_property(prop_value: Union[str, List, Dict, None], prop_type: str) -> str:
    """Gets the dominant simple value (copied/adapted from dashboard/cortex)."""
    if prop_type == 'polarity':
        if isinstance(prop_value, list) and prop_value: return max(prop_value, key=lambda item: item[1])[0]
        elif isinstance(prop_value, str) and prop_value in SIMPLE_POLARITIES: return prop_value
        else: return '?'
    elif prop_type == 'alignment':
        if isinstance(prop_value, dict) and prop_value: return max(prop_value, key=prop_value.get, default='Error')
        elif isinstance(prop_value, str) and prop_value in SIMPLE_ALIGNMENTS: return prop_value
        else: return 'Error'
    return '?'

def _calculate_polarity_distance(p1_val, p2_val) -> float:
    """Calculates distance between two polarities (0=identical, 1=opposite, 0.5=neutral)."""
    p1 = _get_dominant_property(p1_val, 'polarity')
    p2 = _get_dominant_property(p2_val, 'polarity')
    if p1 == p2: return 0.0
    # Opposite polarities (+ vs -) have max distance
    if {p1, p2} == {'+', '-'}: return 1.0
    # Neutral (0) vs non-neutral (+ or -)
    if '0' in [p1, p2] and ('+' in [p1, p2] or '-' in [p1, p2]): return 0.6
    # Oscillating (±) vs others
    if '±' in [p1, p2] and ('0' not in [p1, p2]): return 0.5 # vs +/-
    if '±' in [p1, p2] and '0' in [p1, p2]: return 0.3 # vs 0
    # Unknown (?) vs others
    if '?' in [p1, p2]: return 0.7
    return 0.5 # Default distance for other combinations


def _calculate_alignment_distance(a1_val, a2_val) -> float:
    """Calculates distance between two alignments (0=identical, 1=conflicting, 0.5=neutral)."""
    # Simple version based on dominant alignment - can be enhanced with composite logic
    a1 = _get_dominant_property(a1_val, 'alignment')
    a2 = _get_dominant_property(a2_val, 'alignment')
    if a1 == 'Error' or a2 == 'Error': return 0.8 # High distance if error
    if a1 == a2: return 0.0
    # Define known conflicting pairs (adjust based on symbolic theory)
    conflicts = [{'Celestial', 'Chthonic'}, {'Harmonic', 'Elemental'}] # Example
    if {a1, a2} in conflicts: return 1.0
    # Define synergistic pairs (low distance)
    synergies = [{'Harmonic', 'Celestial'}, {'Elemental', 'Void'}] # Example
    if {a1, a2} in synergies: return 0.2
    return 0.6 # Default distance for different but non-conflicting alignments

def compute_symbolic_distance(g1: GlyphData, g2: GlyphData,
                              tfidf_vectorizer: Optional[TfidfVectorizer] = None,
                              tfidf_matrix: Optional[np.ndarray] = None,
                              glyph_indices: Optional[Dict[str, int]] = None) -> float:
    """
    Computes a weighted distance between two glyphs based on their symbolic properties.
    Distance is normalized between 0 (identical) and 1 (maximally dissimilar).

    Args:
        g1: First glyph dictionary.
        g2: Second glyph dictionary.
        tfidf_vectorizer: Pre-fitted TF-IDF vectorizer for tags (optional).
        tfidf_matrix: Pre-computed TF-IDF matrix (optional).
        glyph_indices: Mapping from glyph ID to matrix index (optional).

    Returns:
        A float representing the symbolic distance.
    """
    total_distance = 0.0
    total_weight = 0.0

    # --- Polarity Distance ---
    pol_dist = _calculate_polarity_distance(g1.get('polarité'), g2.get('polarité'))
    total_distance += pol_dist * WEIGHTS_DISTANCE['polarity']
    total_weight += WEIGHTS_DISTANCE['polarity']

    # --- Alignment Distance ---
    aln_dist = _calculate_alignment_distance(g1.get('alignement'), g2.get('alignement'))
    total_distance += aln_dist * WEIGHTS_DISTANCE['alignment']
    total_weight += WEIGHTS_DISTANCE['alignment']

    # --- Numerical Property Distances (Normalized) ---
    for prop, weight in [('fréquence', WEIGHTS_DISTANCE['frequency']),
                         ('poids', WEIGHTS_DISTANCE['weight']),
                         ('entropy_score', WEIGHTS_DISTANCE['entropy'])]:
        v1 = g1.get(prop, DEFAULT_ATTRS.get(prop))
        v2 = g2.get(prop, DEFAULT_ATTRS.get(prop))
        # Define reasonable min/max for normalization (adjust as needed)
        min_max = {'fréquence': (1, 144), 'poids': (1, 10), 'entropy_score': (0.0, 1.0)}
        norm_dist = abs(_normalize(v1, *min_max[prop]) - _normalize(v2, *min_max[prop]))
        total_distance += norm_dist * weight
        total_weight += weight

    # --- Tag Distance (Using TF-IDF Cosine Similarity -> Distance) ---
    tag_weight = WEIGHTS_DISTANCE['tags']
    if tfidf_vectorizer and tfidf_matrix is not None and glyph_indices and g1['id'] in glyph_indices and g2['id'] in glyph_indices:
        idx1 = glyph_indices[g1['id']]
        idx2 = glyph_indices[g2['id']]
        # Cosine similarity -> distance (1 - similarity)
        # Ensure vectors are reshaped correctly for pairwise calculation
        similarity = cosine_similarity(tfidf_matrix[idx1:idx1+1], tfidf_matrix[idx2:idx2+1])[0][0]
        tag_dist = max(0.0, 1.0 - similarity) # Ensure distance is non-negative
        total_distance += tag_dist * tag_weight
    else:
        # Fallback: Jaccard distance if TF-IDF not provided
        tags1 = set(g1.get('tags', []))
        tags2 = set(g2.get('tags', []))
        intersection = len(tags1.intersection(tags2))
        union = len(tags1.union(tags2))
        jaccard_dist = 1.0 - (intersection / union) if union > 0 else 1.0
        total_distance += jaccard_dist * tag_weight
    total_weight += tag_weight

    # Final weighted average distance
    return total_distance / total_weight if total_weight > 0 else 0.5


def build_glyph_graph(glyphs: List[GlyphData],
                      distance_threshold: float = 0.4,
                      include_similarity_edges: bool = True) -> nx.Graph:
    """
    Builds a NetworkX graph representing the glyphic topology.

    Nodes represent glyphs, with properties as attributes.
    Edges represent explicit relationships (from glyph.relationships)
    and/or implicit similarity (below distance_threshold).

    Args:
        glyphs: List of glyph dictionaries.
        distance_threshold: Connect glyphs with distance below this threshold.
        include_similarity_edges: Whether to compute and add similarity edges.

    Returns:
        A NetworkX graph instance.
    """
    G = nx.Graph()
    if not glyphs: return G

    logger.info(f"Building glyph graph for {len(glyphs)} glyphs...")

    # Pre-compute TF-IDF for tag distance calculation (optional but efficient)
    tfidf_vectorizer = None
    tfidf_matrix = None
    glyph_indices = None
    if include_similarity_edges and len(glyphs) > 1:
        try:
            # Use tags for TF-IDF, join list into space-separated string
            corpus = [" ".join(g.get('tags', [])) for g in glyphs]
            glyph_indices = {g['id']: i for i, g in enumerate(glyphs)}
            tfidf_vectorizer = TfidfVectorizer(lowercase=True, stop_words='english', max_features=500)
            tfidf_matrix = tfidf_vectorizer.fit_transform(corpus)
            logger.debug(f"TF-IDF matrix computed: shape {tfidf_matrix.shape}")
        except Exception as e:
            logger.warning(f"TF-IDF calculation failed: {e}. Falling back to Jaccard distance for tags.")
            tfidf_vectorizer = None; tfidf_matrix = None; glyph_indices = None

    # Add nodes with attributes
    for glyph in glyphs:
        if 'id' not in glyph: continue # Skip invalid glyphs
        # Prepare attributes for the node, flatten complex ones for visualization maybe
        node_attrs = {
            'id': glyph['id'], # Ensure ID is an attribute too
            'timestamp': glyph.get('timestamp'),
            'polarity': _get_dominant_property(glyph.get('polarité'), 'polarity'),
            'alignment': _get_dominant_property(glyph.get('alignement'), 'alignment'),
            'frequency': glyph.get('fréquence', DEFAULT_ATTRS['fréquence']),
            'weight': glyph.get('poids', DEFAULT_ATTRS['poids']),
            'entropy': glyph.get('entropy_score', DEFAULT_ATTRS['entropy_score']),
            'resonance': glyph.get('resonance', DEFAULT_ATTRS['resonance']),
            'emergence': glyph.get('emergence', DEFAULT_ATTRS['emergence']),
            'stability': glyph.get('stability'),
            'status': glyph.get('status'),
            'tags': ", ".join(glyph.get('tags', [])), # Join tags for display
            'composite_pol': is_composite_polarity(glyph.get('polarité')),
            'composite_aln': is_composite_alignment(glyph.get('alignement')),
            'is_fused': 'fused' in glyph.get('tags', []),
            'is_reflexive': 'reflexive' in glyph.get('tags', []),
            # Add more attributes as needed for visualization/analysis
        }
        # Filter out None values before adding node
        node_attrs_clean = {k: v for k, v in node_attrs.items() if v is not None}
        G.add_node(glyph['id'], **node_attrs_clean)

    # Add explicit relationship edges
    explicit_edge_count = 0
    for glyph in glyphs:
        glyph_id = glyph['id']
        if not G.has_node(glyph_id): continue # Skip if node wasn't added
        for rel in glyph.get('relationships', []):
            target_id = rel.get('target_glyph_id')
            rel_type = rel.get('type')
            if target_id and rel_type and G.has_node(target_id):
                 # Add edge with relationship type as attribute
                 # For undirected graph, relationships like 'causes'/'caused_by' are less distinct
                 G.add_edge(glyph_id, target_id, type='explicit', relationship=rel_type, weight=0.8) # Higher weight for explicit
                 explicit_edge_count += 1
    logger.debug(f"Added {explicit_edge_count} explicit relationship edges.")

    # Add implicit similarity edges (if enabled)
    similarity_edge_count = 0
    if include_similarity_edges and len(glyphs) > 1:
        logger.debug(f"Calculating similarity edges (threshold: {distance_threshold})...")
        # Compute pairwise distances (can be slow for large N)
        for i in range(len(glyphs)):
            for j in range(i + 1, len(glyphs)):
                g1 = glyphs[i]
                g2 = glyphs[j]
                if 'id' not in g1 or 'id' not in g2: continue

                distance = compute_symbolic_distance(g1, g2, tfidf_vectorizer, tfidf_matrix, glyph_indices)
                if distance < distance_threshold:
                    # Add edge if distance is below threshold
                    # Weight edge inversely proportional to distance (closer = stronger edge)
                    similarity_weight = max(0.01, 1.0 - (distance / distance_threshold)) # Scale 0-1
                    # Avoid adding edge if an explicit one already exists (or update weight?)
                    if not G.has_edge(g1['id'], g2['id']):
                         G.add_edge(g1['id'], g2['id'], type='similarity', distance=round(distance, 3), weight=round(similarity_weight, 2))
                         similarity_edge_count += 1
        logger.debug(f"Added {similarity_edge_count} similarity edges.")

    logger.info(f"Glyph graph built: {G.number_of_nodes()} nodes, {G.number_of_edges()} edges.")
    return G


def cluster_glyphs(glyphs: List[GlyphData],
                     strategy: str = 'alignment_polarity') -> Dict[str, List[str]]:
    """
    Groups glyphs into clusters based on a chosen strategy.

    Args:
        glyphs: List of glyph dictionaries.
        strategy: Clustering strategy ('alignment', 'polarity', 'alignment_polarity', 'resonance_band').

    Returns:
        A dictionary where keys are cluster names and values are lists of glyph IDs.
    """
    clusters: Dict[str, List[str]] = {}
    if not glyphs: return clusters

    logger.info(f"Clustering {len(glyphs)} glyphs using strategy: {strategy}...")

    for glyph in glyphs:
        if 'id' not in glyph: continue
        cluster_key = "Unclustered"

        try:
            if strategy == 'alignment':
                cluster_key = f"Align: {_get_dominant_property(glyph.get('alignement'), 'alignment')}"
            elif strategy == 'polarity':
                cluster_key = f"Pol: {_get_dominant_property(glyph.get('polarité'), 'polarity')}"
            elif strategy == 'alignment_polarity':
                align = _get_dominant_property(glyph.get('alignement'), 'alignment')
                pol = _get_dominant_property(glyph.get('polarité'), 'polarity')
                cluster_key = f"{align} / {pol}"
            elif strategy == 'resonance_band':
                res = glyph.get('resonance', DEFAULT_ATTRS['resonance'])
                if res >= 75: band = "High-Res"
                elif res >= 40: band = "Mid-Res"
                else: band = "Low-Res"
                cluster_key = f"Res: {band}"
            elif strategy == 'stability':
                 stab = glyph.get('stability', 'MEDIUM')
                 cluster_key = f"Stab: {stab}"

            if cluster_key not in clusters:
                clusters[cluster_key] = []
            clusters[cluster_key].append(glyph['id'])

        except Exception as e:
             logger.warning(f"Failed to assign cluster key for glyph {glyph['id']}: {e}")
             if "Unclustered" not in clusters: clusters["Unclustered"] = []
             clusters["Unclustered"].append(glyph['id'])

    # Optional: Filter out very small clusters
    final_clusters = {k: v for k, v in clusters.items() if len(v) >= MIN_CLUSTER_SIZE}
    unclustered_count = sum(len(v) for k, v in clusters.items() if len(v) < MIN_CLUSTER_SIZE)
    if unclustered_count > 0:
         logger.info(f"Filtered out {len(clusters) - len(final_clusters)} clusters smaller than {MIN_CLUSTER_SIZE} ({unclustered_count} glyphs).")

    logger.info(f"Clustering complete. Found {len(final_clusters)} clusters.")
    return final_clusters

3. Auxiliary Files (topology_utils.py, etc.)

For this phase, the core logic is integrated directly into glyph_topology_engine.py. A separate topology_utils.py isn't strictly necessary yet, but could be created later if helper functions become numerous or complex (e.g., advanced distance metrics, layout algorithms). Summary of Changes and Integration:

glyph_topology_engine.py:

Created with core functions: compute_symbolic_distance (calculates weighted distance based on properties, including TF-IDF for tags), build_glyph_graph (constructs a NetworkX graph with nodes, explicit relationship edges, and implicit similarity edges based on distance), and cluster_glyphs (groups glyphs based on selected strategies).

glyph_dashboard_phase_vii.py:

Builds upon the Phase VI dashboard.

Imports networkx, pyvis, and the new topology engine functions.

Adds the "🕸️ Topology" tab.

Includes controls in the topology tab sidebar for selecting the clustering strategy and similarity distance threshold.

Adds a button to trigger re-clustering (st.session_state.current_clusters stores the result).

Defines render_topology_graph which uses build_glyph_graph and cluster_glyphs to generate data for a Pyvis network visualization. It colors nodes by cluster and adds edges based on relationships/similarity.

Embeds the generated Pyvis HTML graph directly into the Streamlit tab using st.components.v1.html.

Uses st.session_state to maintain the current clustering results and graph layout seed between interactions.

This implementation provides the necessary engine for topological analysis and the visual interface for observing the spatial structure and clusters within the glyphic field.
